5//! CommitSyncer implements efficient synchronization of committed data.
6//!
7//! During the operation of a committee of authorities for consensus, one or
8//! more authorities can fall behind the quorum in their received and accepted
9//! blocks. This can happen due to network disruptions, host crash, or other
10//! reasons. Authorities fell behind need to catch up to the quorum to be able
11//! to vote on the latest leaders. So efficient synchronization is necessary
12//! to minimize the impact of temporary disruptions and maintain smooth
13//! operations of the network.
14//! CommitSyncer achieves efficient synchronization by relying on the following:
15//! when blocks are included in commits with >= 2f+1 certifiers by stake, these
16//! blocks must have passed verifications on some honest validators, so
17//! re-verifying them is unnecessary. In fact, the quorum certified commits
18//! themselves can be trusted to be sent to IOTA directly, but for simplicity
19//! this is not done. Blocks from trusted commits still go through Core and
20//! committer.
21//!
22//! Another way CommitSyncer improves the efficiency of synchronization is
23//! parallel fetching: commits have a simple dependency graph (linear), so it is
24//! easy to fetch ranges of commits in parallel.
25//!
26//! Commit synchronization is an expensive operation, involving transferring
27//! large amount of data via the network. And it is not on the critical path of
28//! block processing. So the heuristics for synchronization, including triggers
29//! and retries, should be chosen to favor throughput and efficient resource
30//! usage, over faster reactions.

81impl CommitSyncType {
82    pub(crate) fn commit_sync_batch_size(&self, context: &Context) -> u32 {
83        match self {
84            CommitSyncType::Fast => context.parameters.fast_commit_sync_batch_size,
85            CommitSyncType::Regular => context.parameters.commit_sync_batch_size,
86        }
87    }
88
89    pub(crate) fn as_str(&self) -> &'static str {
90        match self {
91            CommitSyncType::Fast => "fast_commit_sync",
92            CommitSyncType::Regular => "commit_sync",
93        }
94    }
95
96    pub(crate) fn should_schedule(
97        &self,
98        gap: u32,
99        commit_sync_gap_threshold: u32,
100        consensus_fast_commit_sync: bool,
101    ) -> bool {
102        match self {
103            // Fast syncer requires consensus_fast_commit_sync to be enabled
104            CommitSyncType::Fast => consensus_fast_commit_sync && gap > commit_sync_gap_threshold,
105            // Regular syncer handles all gaps when consensus_fast_commit_sync is disabled,
106            // otherwise only handles small gaps
107            CommitSyncType::Regular => {
108                !consensus_fast_commit_sync || gap <= commit_sync_gap_threshold
109            }
110        }
111    }
112}

114/// Verifies that fetched block headers match the requested block refs.
115/// Returns verified headers or an error if count/reference mismatch.
116pub(crate) fn verify_fetched_headers(
117    peer: AuthorityIndex,
118    request_block_refs: &[BlockRef],
119    serialized_block_headers: Vec<Bytes>,
120) -> ConsensusResult<Vec<VerifiedBlockHeader>> {
121    // 1. Verify count matches
122    if request_block_refs.len() != serialized_block_headers.len() {
123        return Err(ConsensusError::UnexpectedNumberOfHeadersFetched {
124            authority: peer,
125            requested: request_block_refs.len(),
126            received_headers: serialized_block_headers.len(),
127        });
128    }
129
130    // 2. Verify each header's reference matches requested
131    serialized_block_headers
132        .into_iter()
133        .zip(request_block_refs)
134        .map(|(serialized, requested_ref)| {
135            let header = VerifiedBlockHeader::new_from_bytes(serialized)?;
136            if *requested_ref != header.reference() {
137                return Err(ConsensusError::UnexpectedBlockHeaderForCommit {
138                    peer,
139                    requested: *requested_ref,
140                    received: header.reference(),
141                });
142            }
143            Ok(header)
144        })
145        .collect()
146}

148// Handle to stop the CommitSyncer loop.
149pub(crate) struct CommitSyncerHandle {
150    schedule_task: JoinHandle<()>,
151    tx_shutdown: oneshot::Sender<()>,
152}
153
154impl CommitSyncerHandle {
155    pub(crate) async fn stop(self) {
156        let _ = self.tx_shutdown.send(());
157        // Do not abort schedule task, which waits for fetches to shut down.
158        if let Err(e) = self.schedule_task.await {
159            if e.is_panic() {
160                std::panic::resume_unwind(e.into_panic());
161            }
162        }
163    }
164}

223/// Rejects a deserialized commit whose variant does not match the local
224/// protocol-flag configuration. The flag is uniform across the network within
225/// an epoch, so any mismatch implies either a malicious peer or a misconfigured
226/// upgrade path. Called on commits received over the network in
227/// `verify_commits`; recovery from local store is trusted and skips this check.
228pub(crate) fn check_commit_version_matches_flags(
229    commit: &Commit,
230    protocol_config: &iota_protocol_config::ProtocolConfig,
231) -> ConsensusResult<()> {
232    let fast_commit_sync = protocol_config.consensus_fast_commit_sync();
233    let starfish_speed = protocol_config.consensus_starfish_speed();
234    let variant_matches_flags = matches!(
235        (commit, fast_commit_sync, starfish_speed),
236        (Commit::V1(_), false, false) | (Commit::V2(_), true, false) | (Commit::V3(_), true, true)
237    );
238    if !variant_matches_flags {
239        let actual = match commit {
240            Commit::V1(_) => "V1",
241            Commit::V2(_) => "V2",
242            Commit::V3(_) => "V3",
243        };
244        return Err(ConsensusError::WrongCommitVersionForFlags {
245            actual,
246            fast_commit_sync,
247            starfish_speed,
248        });
249    }
250    Ok(())
251}

253/// Validates every `AuthorityIndex` carried by a fetched commit against the
254/// committee, so malformed indices are rejected at ingress with peer
255/// attribution instead of panicking later when commit content is indexed
256/// into per-authority state.
257fn verify_commit_authority_indices(
258    context: &Context,
259    peer: AuthorityIndex,
260    commit: &Commit,
261) -> ConsensusResult<()> {
262    let committee = &context.committee;
263    let check = |index: AuthorityIndex| -> ConsensusResult<()> {
264        if !committee.is_valid_index(index) {
265            return Err(ConsensusError::InvalidAuthorityIndexRequested {
266                index,
267                max: committee.size(),
268                peer,
269            });
270        }
271        Ok(())
272    };
273    check(commit.leader().author)?;
274    for block_ref in commit.block_headers() {
275        check(block_ref.author)?;
276    }
277    for transaction_ref in commit.committed_transactions() {
278        check(transaction_ref.author())?;
279    }
280    for (index, _) in commit.reputation_scores() {
281        check(*index)?;
282    }
283    Ok(())
284}

286/// Free-function form of `Inner::verify_commits`, taking only the inputs the
287/// verification actually uses (`Context` and `BlockVerifier`). Lets tests
288/// exercise the full deserialize-and-verify path without constructing a full
289/// `Inner<C>` fixture.
290pub(crate) fn verify_commits(
291    context: &Arc<Context>,
292    block_verifier: &dyn BlockVerifier,
293    misbehavior_store: &MisbehaviorStore,
294    peer: AuthorityIndex,
295    commit_range: CommitRange,
296    serialized_commits: Vec<Bytes>,
297    serialized_vote_blocks_headers: Vec<Bytes>,
298    max_commits: usize,
299) -> ConsensusResult<(Vec<TrustedCommit>, Vec<VerifiedBlockHeader>)> {
300    // Validate response size - peer should not return more than max_commits
301    if serialized_commits.len() > max_commits {
302        return Err(ConsensusError::TooManyCommitsFromPeer {
303            peer,
304            count: serialized_commits.len() as CommitIndex,
305            limit: max_commits as CommitIndex,
306        });
307    }
308
309    // One vote header per authority certifies a commit, but servers that do
310    // not dedup votes by author may legitimately serve a few more (e.g. an
311    // author re-including its vote in a block after crash recovery), so allow
312    // some multiplicity while still bounding signature verification work.
313    let max_vote_headers = context
314        .committee
315        .size()
316        .saturating_mul(MAX_COMMIT_VOTE_HEADERS_PER_AUTHORITY);
317    if serialized_vote_blocks_headers.len() > max_vote_headers {
318        return Err(ConsensusError::TooManyCommitVoteHeaders {
319            peer,
320            count: serialized_vote_blocks_headers.len(),
321            limit: max_vote_headers,
322        });
323    }
324
325    // Parse and verify commits.
326    let mut commits = Vec::new();
327    for serialized in &serialized_commits {
328        let commit: Commit =
329            bcs::from_bytes(serialized).map_err(ConsensusError::MalformedCommit)?;
330        check_commit_version_matches_flags(&commit, &context.protocol_config)?;
331        verify_commit_authority_indices(context, peer, &commit)?;
332        let digest = TrustedCommit::compute_digest(serialized);
333        if commits.is_empty() {
334            // start is inclusive, so first commit must be at the start index.
335            if commit.index() != commit_range.start() {
336                return Err(ConsensusError::UnexpectedStartCommit {
337                    peer,
338                    start: commit_range.start(),
339                    commit: Box::new(commit),
340                });
341            }
342        } else {
343            // Verify next commit increments index and references the previous digest.
344            let (last_commit_digest, last_commit): &(CommitDigest, Commit) =
345                commits.last().unwrap();
346            if commit.index() != last_commit.index() + 1
347                || &commit.previous_digest() != last_commit_digest
348            {
349                return Err(ConsensusError::UnexpectedCommitSequence {
350                    peer,
351                    prev_commit: Box::new(last_commit.clone()),
352                    curr_commit: Box::new(commit),
353                });
354            }
355        }
356        commits.push((digest, commit));
357    }
358    let Some((end_commit_digest, end_commit)) = commits.last() else {
359        return Err(ConsensusError::NoCommitReceived { peer });
360    };
361
362    // Parse and verify blocks. Then accumulate votes on the end commit.
363    let end_commit_ref = CommitRef::new(end_commit.index(), *end_commit_digest);
364    let mut stake_aggregator = StakeAggregator::<QuorumThreshold>::new();
365    let mut verified_voting_headers = Vec::new();
366    for serialized_block_header in serialized_vote_blocks_headers {
367        let signed_block_header: SignedBlockHeader = bcs::from_bytes(&serialized_block_header)
368            .map_err(ConsensusError::MalformedHeader)
369            .inspect_err(|e| {
370                // Author is unknown when deserialization fails — blame the peer.
371                misbehavior_store.record_faulty_block_header(peer, peer, e);
372            })?;
373        // The block signature needs to be verified.
374        if let Err(e) = block_verifier.verify(&signed_block_header) {
375            misbehavior_store.record_faulty_block_header(peer, signed_block_header.author(), &e);
376            return Err(e);
377        }
378        for vote in signed_block_header.commit_votes() {
379            if *vote == end_commit_ref {
380                stake_aggregator.add(signed_block_header.author(), &context.committee);
381            }
382        }
383        // Store the verified voting block header
384        verified_voting_headers.push(VerifiedBlockHeader::new_verified(
385            signed_block_header,
386            serialized_block_header,
387        ));
388    }
389
390    // Check if the end commit has enough votes.
391    if !stake_aggregator.reached_threshold(&context.committee) {
392        return Err(ConsensusError::NotEnoughCommitVotes {
393            stake: stake_aggregator.stake(),
394            peer,
395            commit: Box::new(end_commit.clone()),
396        });
397    }
398
399    let trusted_commits = commits
400        .into_iter()
401        .zip(serialized_commits)
402        .map(|((_d, c), s)| TrustedCommit::new_trusted(c, s))
403        .collect();
404    Ok((trusted_commits, verified_voting_headers))
405}

407/// Verifies transactions against their block headers and returns a map of
408/// BlockRef to VerifiedTransactions.
409pub(crate) fn verify_transactions_with_headers(
410    context: Arc<Context>,
411    peer: AuthorityIndex,
412    serialized_transactions: BTreeMap<GenericTransactionRef, Bytes>,
413    block_headers: BTreeMap<BlockRef, VerifiedBlockHeader>,
414) -> ConsensusResult<BTreeMap<GenericTransactionRef, VerifiedTransactions>> {
415    let mut verified_transactions_map = BTreeMap::new();
416    let mut encoder = create_encoder(&context);
417    let size_limit = serialized_transactions_size_limit(&context);
418    for (committed_transactions_ref, inner_serialized_transactions) in serialized_transactions {
419        let block_ref = match committed_transactions_ref {
420            GenericTransactionRef::BlockRef(br) => br,
421            _ => {
422                return Err(ConsensusError::TransactionRefVariantMismatch {
423                    protocol_flag_enabled: false,
424                    expected_variant: "BlockRef",
425                    received_variant: "TransactionRef",
426                });
427            }
428        };
429        // Bound the peer-supplied payload before erasure-encoding it.
430        if inner_serialized_transactions.len() > size_limit {
431            return Err(ConsensusError::SerializedTransactionsTooLarge {
432                size: inner_serialized_transactions.len(),
433                limit: size_limit,
434            });
435        }
436        // Step 1: Get the block header and verify that the transactions commitment
437        // matches. This ensures the transactions we received are exactly
438        // the ones that were included in the block when it was created.
439        let block_header = block_headers
440            .get(&block_ref)
441            .ok_or(ConsensusError::MissingBlockHeader { block_ref })?;
442
443        if block_header.transactions_commitment()
444            != TransactionsCommitment::compute_transactions_commitment(
445                &inner_serialized_transactions,
446                &context,
447                &mut encoder,
448            )?
449        {
450            return Err(ConsensusError::TransactionCommitmentFailure {
451                round: block_ref.round,
452                author: block_ref.author,
453                peer,
454            });
455        }
456
457        // Step 2: Deserialize the actual transactions vector.
458        let transactions: Vec<Transaction> = bcs::from_bytes(&inner_serialized_transactions)
459            .map_err(ConsensusError::MalformedTransactions)?;
460
461        // Step 3: Create a VerifiedTransactions instance and insert into map
462        let verified_transactions = VerifiedTransactions::new(
463            transactions,
464            TransactionRef::new(block_ref, block_header.transactions_commitment()),
465            Some(block_ref.digest),
466            inner_serialized_transactions,
467        );
468
469        verified_transactions_map.insert(
470            GenericTransactionRef::BlockRef(block_ref),
471            verified_transactions,
472        );
473    }
474
475    Ok(verified_transactions_map)
476}

478/// Verifies transactions against their transaction refs and returns a map of
479/// BlockRef to VerifiedTransactions.
480pub(crate) fn verify_transactions_with_transactions_refs(
481    context: &Arc<Context>,
482    peer: AuthorityIndex,
483    serialized_transactions: BTreeMap<GenericTransactionRef, Bytes>,
484) -> ConsensusResult<BTreeMap<GenericTransactionRef, VerifiedTransactions>> {
485    let mut verified_transactions_map = BTreeMap::new();
486    let mut encoder = create_encoder(context);
487    let size_limit = serialized_transactions_size_limit(context);
488    for (committed_transactions_ref, inner_serialized_transactions) in serialized_transactions {
489        let transaction_ref = match committed_transactions_ref {
490            GenericTransactionRef::TransactionRef(tx_ref) => tx_ref,
491            _ => {
492                return Err(ConsensusError::TransactionRefVariantMismatch {
493                    protocol_flag_enabled: true,
494                    expected_variant: "TransactionRef",
495                    received_variant: "BlockRef",
496                });
497            }
498        };
499        // Bound the peer-supplied payload before erasure-encoding it.
500        if inner_serialized_transactions.len() > size_limit {
501            return Err(ConsensusError::SerializedTransactionsTooLarge {
502                size: inner_serialized_transactions.len(),
503                limit: size_limit,
504            });
505        }
506        // Step 1: Verify that the transaction commitment matches.
507        if transaction_ref.transactions_commitment
508            != TransactionsCommitment::compute_transactions_commitment(
509                &inner_serialized_transactions,
510                context,
511                &mut encoder,
512            )?
513        {
514            return Err(ConsensusError::TransactionCommitmentFailure {
515                round: transaction_ref.round,
516                author: transaction_ref.author,
517                peer,
518            });
519        }
520
521        // Step 2: Deserialize the actual transactions vector.
522        let transactions: Vec<Transaction> = bcs::from_bytes(&inner_serialized_transactions)
523            .map_err(ConsensusError::MalformedTransactions)?;
524
525        // Step 3: Create a VerifiedTransactions instance and insert into map
526        let verified_transactions = VerifiedTransactions::new(
527            transactions,
528            transaction_ref,
529            None,
530            inner_serialized_transactions,
531        );
532
533        verified_transactions_map.insert(
534            GenericTransactionRef::TransactionRef(transaction_ref),
535            verified_transactions,
536        );
537    }
538
539    Ok(verified_transactions_map)
540}

542/// Generic fetch loop that retries fetching data from available authorities
543/// until a request succeeds. This is shared between RegularCommitSyncer and
544/// FastCommitSyncer.
545///
546/// # Type Parameters
547/// - `C`: Network client type
548/// - `T`: Fetched data type (CertifiedCommits for regular, (Vec<TrustedCommit>,
549///   Vec<CommittedSubDag>) for fast)
550/// - `F`: Fetch function type
551/// - `Fut`: Future returned by fetch function
552///
553/// # Parameters
554/// - `inner`: Shared context and dependencies
555/// - `commit_range`: The range of commits to fetch
556/// - `fetch_timeout_multiplier`: Multiplier for timeout calculation (4 for
557///   regular, 2 for fast)
558/// - `fetch_once_fn`: Implementation-specific fetch function
559///
560/// # Returns
561/// Tuple of (end_commit_index, fetched_data)
562#[cfg_attr(test, tracing::instrument(skip_all, fields(authority = %inner.context.own_index)))]
563pub(crate) async fn fetch_loop<C, T, F, Fut>(
564    inner: Arc<Inner<C>>,
565    commit_range: CommitRange,
566    fetch_timeout_multiplier: u32,
567    fetch_once_fn: F,
568) -> (CommitIndex, T)
569where
570    C: NetworkClient,
571    T: Send,
572    F: Fn(Arc<Inner<C>>, AuthorityIndex, CommitRange, Duration) -> Fut,
573    Fut: std::future::Future<Output = ConsensusResult<T>> + Send,
574{
575    // Individual request base timeout.
576    #[cfg(not(test))]
577    const TIMEOUT: Duration = Duration::from_secs(10);
578    #[cfg(test)]
579    const TIMEOUT: Duration = Duration::from_millis(500);
580    // Max per-request timeout will be base timeout times a multiplier.
581    // At the extreme, this means there will be 120s timeout to fetch
582    // max_headers_per_commit_sync_fetch headers.
583    const MAX_TIMEOUT_MULTIPLIER: u32 = 12;
584    // timeout * max number of targets should be reasonably small, so the
585    // system can adjust to slow network or large data sizes quickly.
586    const MAX_NUM_TARGETS: usize = 24;
587    let mut timeout_multiplier = 0;
588
589    let _timer = inner
590        .context
591        .metrics
592        .node_metrics
593        .commit_sync_fetch_loop_latency
594        .start_timer();
595    info!(
596        "[{}] Starting to fetch commits in {commit_range:?} ...",
597        inner.sync_type.as_str()
598    );
599    loop {
600        // Attempt to fetch commits and blocks through min(committee size,
601        // MAX_NUM_TARGETS) peers.
602        let mut target_authorities = inner
603            .context
604            .committee
605            .authorities()
606            .filter_map(|(i, _)| {
607                if i != inner.context.own_index {
608                    Some(i)
609                } else {
610                    None
611                }
612            })
613            .collect_vec();
614        #[cfg(not(test))]
615        target_authorities.shuffle(&mut ThreadRng::default());
616        target_authorities.truncate(MAX_NUM_TARGETS);
617        // Increase timeout multiplier for each loop until MAX_TIMEOUT_MULTIPLIER.
618        timeout_multiplier = (timeout_multiplier + 1).min(MAX_TIMEOUT_MULTIPLIER);
619        let request_timeout = TIMEOUT * timeout_multiplier;
620
621        let fetch_timeout = request_timeout * fetch_timeout_multiplier;
622        // Try fetching from the selected target authority.
623        for authority in target_authorities {
624            match tokio::time::timeout(
625                fetch_timeout,
626                fetch_once_fn(
627                    inner.clone(),
628                    authority,
629                    commit_range.clone(),
630                    request_timeout,
631                ),
632            )
633            .await
634            {
635                Ok(Ok(data)) => {
636                    info!(
637                        "[{}] Finished fetching commits in {commit_range:?}",
638                        inner.sync_type.as_str()
639                    );
640                    return (commit_range.end(), data);
641                }
642                Ok(Err(e)) => {
643                    let hostname = inner
644                        .context
645                        .committee
646                        .authority(authority)
647                        .hostname
648                        .clone();
649                    warn!(
650                        "[{}] Failed to fetch {commit_range:?} from {hostname}: {}",
651                        inner.sync_type.as_str(),
652                        e
653                    );
654                    let error: &'static str = e.into();
655                    inner
656                        .context
657                        .metrics
658                        .node_metrics
659                        .commit_sync_fetch_once_errors
660                        .with_label_values(&[hostname.as_str(), error, inner.sync_type.as_str()])
661                        .inc();
662                }
663                Err(_) => {
664                    let hostname = inner
665                        .context
666                        .committee
667                        .authority(authority)
668                        .hostname
669                        .clone();
670                    warn!(
671                        "[{}] Timed out fetching {commit_range:?} from {authority}",
672                        inner.sync_type.as_str()
673                    );
674                    inner
675                        .context
676                        .metrics
677                        .node_metrics
678                        .commit_sync_fetch_once_errors
679                        .with_label_values(&[
680                            hostname.as_str(),
681                            "FetchTimeout",
682                            inner.sync_type.as_str(),
683                        ])
684                        .inc();
685                }
686            }
687        }
688        // Avoid busy looping, by waiting for a while before retrying.
689        sleep(TIMEOUT).await;
690    }
691}

693/// Generic function to start pending fetches while respecting parallelism
694/// limits. This is shared between RegularCommitSyncer and FastCommitSyncer.
695///
696/// # Parameters
697/// - `inner`: Shared context and dependencies
698/// - `pending_fetches`: Set of commit ranges pending fetch
699/// - `fetched_ranges_count`: Number of fetched ranges waiting to be processed
700/// - `inflight_fetches_count`: Number of currently in-flight fetch tasks
701/// - `synced_commit_index`: Latest synced commit index
702/// - `spawn_fn`: Closure to spawn a new fetch task
703///
704/// # Returns
705/// Updated counts after spawning new fetches
706pub(crate) fn try_start_fetches<C, F>(
707    inner: &Arc<Inner<C>>,
708    pending_fetches: &mut BTreeSet<CommitRange>,
709    fetched_ranges_count: usize,
710    inflight_fetches_count: usize,
711    synced_commit_index: CommitIndex,
712    mut spawn_fn: F,
713) -> (usize, usize)
714where
715    C: NetworkClient,
716    F: FnMut(CommitRange),
717{
718    // Cap parallel fetches based on configured limit and committee size, to avoid
719    // overloading the network. Also when there are too many fetched block headers
720    // that cannot be sent to Core before an earlier fetch has not finished,
721    // reduce parallelism so the earlier fetch can retry on a better host and
722    // succeed.
723    let target_parallel_fetches = inner
724        .context
725        .parameters
726        .commit_sync_parallel_fetches
727        .min(inner.context.committee.size() * 2 / 3)
728        .min(
729            inner
730                .context
731                .parameters
732                .commit_sync_batches_ahead
733                .saturating_sub(fetched_ranges_count),
734        )
735        .max(1);
736
737    let mut new_inflight_count = inflight_fetches_count;
738
739    // Start new fetches if there are pending batches and available slots.
740    loop {
741        if new_inflight_count >= target_parallel_fetches {
742            break;
743        }
744        if !pending_fetches.is_empty() {
745            info!(
746                "[{}] Pending fetches: {:?}, target parallel fetches: {}, inflight fetch number: {}",
747                inner.sync_type.as_str(),
748                pending_fetches,
749                target_parallel_fetches,
750                new_inflight_count
751            );
752        }
753        let Some(commit_range) = pending_fetches.pop_first() else {
754            break;
755        };
756        spawn_fn(commit_range);
757        new_inflight_count += 1;
758    }
759
760    let metrics = &inner.context.metrics.node_metrics;
761    let sync_label = inner.sync_type.as_str();
762    metrics
763        .commit_sync_inflight_fetches
764        .with_label_values(&[sync_label])
765        .set(new_inflight_count as i64);
766    metrics
767        .commit_sync_pending_fetches
768        .with_label_values(&[sync_label])
769        .set(pending_fetches.len() as i64);
770    metrics
771        .commit_sync_highest_synced_index
772        .with_label_values(&[sync_label])
773        .set(synced_commit_index as i64);
774
775    (new_inflight_count, pending_fetches.len())
776}

782/// Result of scheduling commit range fetches.
783pub(crate) struct ScheduleResult {
784    /// The new highest scheduled commit index (if any ranges were scheduled).
785    pub new_highest_scheduled: Option<CommitIndex>,
786    /// The commit ranges that were scheduled for fetching.
787    pub ranges_scheduled: Vec<CommitRange>,
788}
789
790/// Creates commit range batches for fetching, respecting backpressure.
791///
792/// This function is shared between RegularCommitSyncer and FastCommitSyncer.
793/// It calculates which commit ranges should be fetched next based on:
794/// - The current gap between local and quorum commit indices
795/// - Backpressure from unhandled commits
796///
797/// # Parameters
798/// - `inner`: Shared context and dependencies
799/// - `fetch_after_index`: Start fetching from commits after this index
800/// - `quorum_commit_index`: The commit index that quorum has reached
801/// - `highest_handled_index`: The highest commit index that has been processed
802/// - `unhandled_commits_threshold`: Threshold for applying backpressure
803///
804/// # Returns
805/// A `ScheduleResult` with the new highest scheduled index and ranges to fetch.
806pub(crate) fn schedule_commit_ranges<C: NetworkClient>(
807    inner: &Inner<C>,
808    fetch_after_index: CommitIndex,
809    quorum_commit_index: CommitIndex,
810    highest_handled_index: CommitIndex,
811    unhandled_commits_threshold: CommitIndex,
812) -> ScheduleResult {
813    let step = inner.sync_type.commit_sync_batch_size(&inner.context);
814    let mut result = ScheduleResult {
815        new_highest_scheduled: None,
816        ranges_scheduled: Vec::new(),
817    };
818
819    for prev_end in (fetch_after_index..=quorum_commit_index).step_by(step as usize) {
820        let range_start = prev_end + 1;
821        let range_end = prev_end + step;
822
823        // Don't schedule incomplete batches
824        if quorum_commit_index < range_end {
825            break;
826        }
827
828        // Apply backpressure if handler is lagging
829        if highest_handled_index + unhandled_commits_threshold < range_end {
830            warn!(
831                "[{}] Skip scheduling new commit fetches: handler lagging. \
832                 highest_handled={}, threshold={}",
833                inner.sync_type.as_str(),
834                highest_handled_index,
835                unhandled_commits_threshold
836            );
837            break;
838        }
839
840        result
841            .ranges_scheduled
842            .push((range_start..=range_end).into());
843        result.new_highest_scheduled = Some(range_end);
844    }
845
846    result
847}

849/// Handles JoinError from fetch tasks.
850///
851/// # Returns
852/// `true` if the syncer should shutdown, `false` otherwise.
853pub(crate) fn handle_fetch_join_error(
854    error: &tokio::task::JoinError,
855    sync_type: &CommitSyncType,
856) -> bool {
857    if error.is_panic() {
858        // Re-panic in the main task
859        return true;
860    }
861    warn!(
862        "[{}] Fetch cancelled. Shutting down: {}",
863        sync_type.as_str(),
864        error
865    );
866    true // Signal to shutdown
867}

869/// Re-queues unfetched portion of a commit range for retry.
870///
871/// When a fetch returns partial results (fewer commits than requested),
872/// this function queues the remaining range for another fetch attempt.
873pub(crate) fn requeue_partial_range(
874    pending_fetches: &mut BTreeSet<CommitRange>,
875    commit_end: CommitIndex,
876    target_end: CommitIndex,
877) {
878    if commit_end < target_end {
879        pending_fetches.insert((commit_end + 1..=target_end).into());
880    }
881}
