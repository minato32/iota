20/// Per-authority misbehavior counters.
21///
22/// Two buckets track different lifecycle stages:
23/// - `in_memory`: from blocks currently in the DAG cache (volatile, recomputed
24///   on each flush from blocks still in cache).
25/// - `persisted`: cumulative counts from blocks evicted from cache (restored
26///   from storage on restart).
27pub(crate) struct MisbehaviorStore {
28    in_memory: CommitteeMisbehaviorCounts,
29    persisted: CommitteeMisbehaviorCounts,
30}
31
32impl MisbehaviorStore {
33    pub(crate) fn new(context: &Context) -> Self {
34        let metrics = &context.metrics.node_metrics;
35        Self {
36            in_memory: CommitteeMisbehaviorCounts::new(&context.committee, metrics, "in_memory"),
37            persisted: CommitteeMisbehaviorCounts::new(&context.committee, metrics, "persisted"),
38        }
39    }
40
41    /// Resets all counters to zero. Used during fast sync reinitialization.
42    pub(crate) fn reset(&self) {
43        self.in_memory.reset();
44        self.persisted.reset();
45    }

47    /// Restores persisted counts from storage and computes in-memory counts
48    /// from the block refs already loaded into the DAG cache.
49    pub(crate) fn initialize_misbehavior_counts(
50        &self,
51        recovered: BTreeMap<AuthorityIndex, MisbehaviorCounts>,
52        recent_refs_by_authority: &[BTreeSet<BlockRef>],
53        evicted_rounds: &[Round],
54        threshold_clock_round: Round,
55        context: &Arc<Context>,
56    ) {
57        for (authority_index, _) in context.committee.authorities() {
58            let idx = authority_index.value();
59
60            // Restore persisted counts from storage. Match exhaustively so a
61            // future MisbehaviorCounts variant trips the compiler here.
62            let storage_metrics = recovered.get(&authority_index).cloned().unwrap_or_default();
63            match storage_metrics {
64                MisbehaviorCounts::V1(inner) => {
65                    self.persisted.set_block_faults(
66                        idx,
67                        inner.faulty_blocks_provable,
68                        inner.faulty_blocks_unprovable,
69                    );
70                    self.persisted.set_dag_faults(
71                        idx,
72                        inner.missing_proposals,
73                        inner.equivocations,
74                    );
75                }
76            }
77
78            // Compute in-memory counts from cached block refs.
79            let eviction_round = evicted_rounds[idx];
80            if threshold_clock_round > 0 {
81                let cached_rounds: Vec<Round> = recent_refs_by_authority[idx]
82                    .iter()
83                    .map(|r| r.round)
84                    .collect();
85                let (eq, missing) = calculate_misbehavior_counts_for_range(
86                    cached_rounds,
87                    eviction_round + 1,
88                    threshold_clock_round.saturating_sub(1),
89                );
90                self.in_memory.set_dag_faults(idx, missing, eq);
91            }
92        }
93    }
94
95    /// Updates misbehavior counts for one authority during flush.
96    /// Must be called before cache eviction is applied, while evicted block
97    /// refs are still in `recent_refs`.
98    /// Returns `Some(MisbehaviorCounts)` if persisted data changed and needs
99    /// to be written to storage.
100    pub(crate) fn update_misbehavior_counts_on_eviction(
101        &self,
102        authority_index: AuthorityIndex,
103        recent_refs: &BTreeSet<BlockRef>,
104        eviction_round: Round,
105        last_eviction_round: Round,
106        threshold_clock_round: Round,
107        context: &Arc<Context>,
108    ) -> Option<MisbehaviorCounts> {
109        if threshold_clock_round == 0 || authority_index.value() >= context.committee.size() {
110            return None;
111        }
112        let idx = authority_index.value();
113
114        // Move buffered faulty block header counts to persisted.
115        let had_faulty = self.flush_faulty_block_header_buffer(idx);
116
117        // Recompute in-memory window from blocks still in cache.
118        let in_memory_block_rounds: Vec<Round> = recent_refs
119            .iter()
120            .map(|b| b.round)
121            .filter(|&r| r > eviction_round && r < threshold_clock_round)
122            .collect();
123        let (in_memory_eq, in_memory_missing) = calculate_misbehavior_counts_for_range(
124            in_memory_block_rounds,
125            eviction_round + 1,
126            threshold_clock_round.saturating_sub(1),
127        );
128        self.in_memory
129            .set_dag_faults(idx, in_memory_missing, in_memory_eq);
130
131        let eviction_advanced = eviction_round != last_eviction_round;
132        if eviction_advanced {
133            // Accumulate newly-evicted rounds into persisted.
134            let evicted_block_rounds: Vec<Round> = recent_refs
135                .iter()
136                .map(|b| b.round)
137                .filter(|&r| r <= eviction_round)
138                .collect();
139            let (evicted_eq, evicted_missing) = calculate_misbehavior_counts_for_range(
140                evicted_block_rounds,
141                last_eviction_round + 1,
142                eviction_round,
143            );
144            self.persisted
145                .add_dag_faults(idx, evicted_missing, evicted_eq);
146        }
147
148        if eviction_advanced || had_faulty {
149            Some(MisbehaviorCounts::V1(self.persisted.snapshot(idx)))
150        } else {
151            None
152        }
153    }
154
155    /// Flush buffered faulty block header counts from in_memory to persisted
156    /// for one authority. Returns true if any counts were moved.
157    fn flush_faulty_block_header_buffer(&self, idx: usize) -> bool {
158        let (prov, unprov) = self.in_memory.drain_block_faults(idx);
159        if prov == 0 && unprov == 0 {
160            return false;
161        }
162        self.persisted.add_block_faults(idx, prov, unprov);
163        true
164    }

166    /// Returns an absolute per-authority snapshot of `persisted + in_memory`
167    /// counts for emission with `CommittedSubDag`. Locks the two buckets
168    /// independently per authority; callers must hold `dag_state.read()` so
169    /// concurrent flush (which writes both buckets under `dag_state.write()`)
170    /// is excluded.
171    pub(crate) fn snapshot_totals(&self) -> Vec<MisbehaviorCounts> {
172        (0..self.in_memory.authorities.len())
173            .map(|i| {
174                let persisted = self.persisted.snapshot(i);
175                let in_memory = self.in_memory.snapshot(i);
176                MisbehaviorCounts::V1(MisbehaviorCountsV1 {
177                    faulty_blocks_provable: persisted.faulty_blocks_provable
178                        + in_memory.faulty_blocks_provable,
179                    faulty_blocks_unprovable: persisted.faulty_blocks_unprovable
180                        + in_memory.faulty_blocks_unprovable,
181                    missing_proposals: persisted.missing_proposals + in_memory.missing_proposals,
182                    equivocations: persisted.equivocations + in_memory.equivocations,
183                })
184            })
185            .collect()
186    }

188    /// Records a faulty block header event detected during block header
189    /// validation. Events are buffered in the in_memory bucket and moved
190    /// to persisted on the next flush.
191    ///
192    /// `peer` is the authority that sent us the block (always known from the
193    /// network connection). `author` is the claimed block author (from the
194    /// header's author field, only trustworthy if the signature is valid).
195    ///
196    /// Attribution rules:
197    /// - Provable faults (valid signature, protocol violation): charged to
198    ///   `author` — cryptographic proof they produced an invalid block. If
199    ///   `peer != author`, the peer is also charged (unprovable) for
200    ///   distributing a block they could have verified themselves.
201    /// - Unprovable faults (bad/missing signature): charged to `peer` only — we
202    ///   can't verify the author field, but we know who sent it to us.
203    pub(crate) fn record_faulty_block_header(
204        &self,
205        peer: AuthorityIndex,
206        author: AuthorityIndex,
207        error: &ConsensusError,
208    ) {
209        // `peer` is trusted (authenticated connection); `author` is
210        // attacker-controlled and may be out of committee range.
211        let committee_size = self.in_memory.authorities.len();
212        let peer_idx = peer.value();
213        let author_idx = author.value();
214        if peer_idx >= committee_size {
215            return;
216        }
217        match classify_block_header_error(error) {
218            FaultType::Provable => {
219                if author_idx >= committee_size {
220                    // Can't credit a bogus author; charge the serving peer instead.
221                    self.in_memory.record_block_fault_unprovable(peer_idx);
222                    return;
223                }
224                self.in_memory.record_block_fault_provable(author_idx);
225                if peer != author {
226                    self.in_memory.record_block_fault_unprovable(peer_idx);
227                }
228            }
229            FaultType::Unprovable => {
230                self.in_memory.record_block_fault_unprovable(peer_idx);
231            }
232            FaultType::Untracked => {}
233        }
234    }
235}

237/// Whether a block header fault can be cryptographically proven.
238enum FaultType {
239    /// Block has a valid author signature but violates protocol rules.
240    /// The signed block header itself is proof of misbehavior.
241    Provable,
242    /// Can't prove authorship — either because the signature is bad or
243    /// missing, or because the header is rejected by a pre-signature check
244    /// (epoch / genesis / author-vs-peer mismatch) so its `author` field
245    /// can't be trusted. Charged to the sending peer, not the claimed author.
246    Unprovable,
247    /// Not counted as misbehavior.
248    Untracked,
249}
250
251fn classify_block_header_error(error: &ConsensusError) -> FaultType {
252    match error {
253        // Pre-signature / parsing errors — the header's author field can't
254        // be trusted, so charge the sender, not the claimed author.
255        ConsensusError::WrongEpoch { .. }
256        | ConsensusError::UnexpectedGenesisHeader
257        | ConsensusError::UnexpectedAuthority(..)
258        | ConsensusError::InvalidAuthorityIndex { .. }
259        | ConsensusError::MalformedHeader(_)
260        | ConsensusError::MalformedTransactions(_)
261        | ConsensusError::MalformedSignature(_)
262        | ConsensusError::SignatureVerificationFailure(_)
263        | ConsensusError::SerializationFailure(_)
264        | ConsensusError::DeserializationFailure(_)
265        | ConsensusError::SerializedTransactionsTooLarge { .. }
266        | ConsensusError::TransactionCommitmentFailure { .. } => FaultType::Unprovable,
267
268        // Signed block header verification — provably the author's fault
269        ConsensusError::TooManyAncestors(..)
270        | ConsensusError::InsufficientParentStakes { .. }
271        | ConsensusError::InvalidAncestorPosition { .. }
272        | ConsensusError::InvalidAncestorRound { .. }
273        | ConsensusError::InvalidGenesisAncestor(_)
274        | ConsensusError::DuplicatedAncestorsAuthority(_)
275        | ConsensusError::InvalidOverlapIndices { .. }
276        | ConsensusError::TransactionTooLarge { .. }
277        | ConsensusError::TooManyTransactions { .. }
278        | ConsensusError::TooManyTransactionBytes { .. }
279        | ConsensusError::InvalidTransaction(_) => FaultType::Provable,
280
281        _ => FaultType::Untracked,
282    }
283}

285/// The four Prometheus gauges that mirror `MisbehaviorCountsV1` for one
286/// `(authority, source)` pair, pre-resolved at construction so per-call
287/// methods don't repeat `with_label_values` lookups and don't have to
288/// thread `&NodeMetrics` and `hostname` through every call site.
289struct AuthorityGauges {
290    block_fault_provable: IntGauge,
291    block_fault_unprovable: IntGauge,
292    missing_proposals: IntGauge,
293    equivocations: IntGauge,
294}
295
296/// Per-authority misbehavior counters and matching Prometheus gauges.
297/// State mutations and gauge updates always happen together inside the same
298/// per-authority `Mutex` lock, so concurrent observers see counter and gauge
299/// agree.
300struct CommitteeMisbehaviorCounts {
301    authorities: Vec<Mutex<MisbehaviorCountsV1>>,
302    gauges: Vec<AuthorityGauges>,
303}
304
305impl CommitteeMisbehaviorCounts {
306    /// `source` is the `source` label baked into all gauge updates for this
307    /// instance (`"in_memory"` or `"persisted"`).
308    fn new(committee: &Committee, metrics: &NodeMetrics, source: &'static str) -> Self {
309        let gauges = committee
310            .authorities()
311            .map(|(_, a)| {
312                let labels = &[a.hostname.as_str(), source];
313                AuthorityGauges {
314                    block_fault_provable: metrics
315                        .faulty_blocks_provable_by_authority
316                        .with_label_values(labels),
317                    block_fault_unprovable: metrics
318                        .faulty_blocks_unprovable_by_peer
319                        .with_label_values(labels),
320                    missing_proposals: metrics
321                        .missing_proposals_by_authority
322                        .with_label_values(labels),
323                    equivocations: metrics.equivocations_by_authority.with_label_values(labels),
324                }
325            })
326            .collect();
327        Self {
328            authorities: (0..committee.size())
329                .map(|_| Mutex::new(MisbehaviorCountsV1::default()))
330                .collect(),
331            gauges,
332        }
333    }
334
335    fn record_block_fault_provable(&self, idx: usize) {
336        let mut c = self.authorities[idx].lock().unwrap();
337        c.faulty_blocks_provable += 1;
338        self.gauges[idx].block_fault_provable.inc();
339    }
340
341    fn record_block_fault_unprovable(&self, idx: usize) {
342        let mut c = self.authorities[idx].lock().unwrap();
343        c.faulty_blocks_unprovable += 1;
344        self.gauges[idx].block_fault_unprovable.inc();
345    }
346
347    /// Atomically take both block-fault counters AND zero the matching gauges.
348    /// The caller transfers the returned `(provable, unprovable)` into the
349    /// persisted bucket via `add_block_faults`.
350    fn drain_block_faults(&self, idx: usize) -> (u64, u64) {
351        let mut c = self.authorities[idx].lock().unwrap();
352        let prov = std::mem::take(&mut c.faulty_blocks_provable);
353        let unprov = std::mem::take(&mut c.faulty_blocks_unprovable);
354        self.gauges[idx].block_fault_provable.set(0);
355        self.gauges[idx].block_fault_unprovable.set(0);
356        (prov, unprov)
357    }
358
359    /// Receive a `drain_block_faults` payload — additive on counters and
360    /// gauges.
361    fn add_block_faults(&self, idx: usize, prov: u64, unprov: u64) {
362        let mut c = self.authorities[idx].lock().unwrap();
363        c.faulty_blocks_provable += prov;
364        c.faulty_blocks_unprovable += unprov;
365        self.gauges[idx].block_fault_provable.add(prov as i64);
366        self.gauges[idx].block_fault_unprovable.add(unprov as i64);
367    }
368
369    /// Overwrite both block-fault counters and gauges. Used to restore from
370    /// storage on startup.
371    fn set_block_faults(&self, idx: usize, prov: u64, unprov: u64) {
372        let mut c = self.authorities[idx].lock().unwrap();
373        c.faulty_blocks_provable = prov;
374        c.faulty_blocks_unprovable = unprov;
375        self.gauges[idx].block_fault_provable.set(prov as i64);
376        self.gauges[idx].block_fault_unprovable.set(unprov as i64);
377    }
378
379    /// Overwrite the DAG-observed faults (`missing_proposals` +
380    /// `equivocations`) and their gauges. Used on `in_memory` where the
381    /// window is recomputed wholesale on every flush.
382    fn set_dag_faults(&self, idx: usize, missing: u64, equiv: u64) {
383        let mut c = self.authorities[idx].lock().unwrap();
384        c.missing_proposals = missing;
385        c.equivocations = equiv;
386        self.gauges[idx].missing_proposals.set(missing as i64);
387        self.gauges[idx].equivocations.set(equiv as i64);
388    }
389
390    /// Accumulate DAG-observed faults — additive on counters and gauges.
391    /// Used on `persisted` to fold in newly-evicted rounds.
392    fn add_dag_faults(&self, idx: usize, missing: u64, equiv: u64) {
393        let mut c = self.authorities[idx].lock().unwrap();
394        c.missing_proposals += missing;
395        c.equivocations += equiv;
396        self.gauges[idx].missing_proposals.add(missing as i64);
397        self.gauges[idx].equivocations.add(equiv as i64);
398    }
399
400    /// Stable read clone of one authority's counters.
401    fn snapshot(&self, idx: usize) -> MisbehaviorCountsV1 {
402        self.authorities[idx].lock().unwrap().clone()
403    }
404
405    fn reset(&self) {
406        for (m, g) in self.authorities.iter().zip(self.gauges.iter()) {
407            let mut c = m.lock().unwrap();
408            *c = MisbehaviorCountsV1::default();
409            g.block_fault_provable.set(0);
410            g.block_fault_unprovable.set(0);
411            g.missing_proposals.set(0);
412            g.equivocations.set(0);
413        }
414    }
415
416    #[cfg(test)]
417    fn collect<F: Fn(&MisbehaviorCountsV1) -> u64>(&self, field: F) -> Vec<u64> {
418        self.authorities
419            .iter()
420            .map(|m| field(&m.lock().unwrap()))
421            .collect()
422    }
423}

425/// Given block rounds for one authority in [start, end], returns
426/// (equivocations, missing_proposals).
427fn calculate_misbehavior_counts_for_range(
428    mut block_rounds: Vec<Round>,
429    start: Round,
430    end: Round,
431) -> (u64, u64) {
432    block_rounds.retain(|&round| round >= start && round <= end);
433    block_rounds.sort();
434    let number_of_blocks = block_rounds.len();
435    block_rounds.dedup();
436    let unique_block_rounds = block_rounds.len();
437    let number_of_equivocations = number_of_blocks.saturating_sub(unique_block_rounds) as u64;
438    let number_of_missing_blocks =
439        (end + 1).saturating_sub(start + unique_block_rounds as u32) as u64;
440    (number_of_equivocations, number_of_missing_blocks)
441}

605            &context,
606        );
607        assert!(result.is_none());
608
609        // No eviction (eviction_round == last_eviction_round) AND no buffered
610        // faulty headers → None. (Eviction advance OR a faulty flush is what
611        // triggers a write; here neither happens.)
612        let result = store.update_misbehavior_counts_on_eviction(

614            &recent_refs,
615            5,
616            5,

618            &context,

629            &context,

641    async fn test_metrics_flush_and_recovery() {
642        let committee_size = 4;
643        let (context, _) = Context::new_for_test(committee_size);
644        let context = context.with_parameters(Parameters {
645            dag_state_cached_rounds: 5,
646            ..Default::default()
647        });
648        let context = Arc::new(context);
649        let store = Arc::new(MemStore::new());
650        let mut dag_state = DagState::new(context.clone(), store.clone());
651
652        // Build a 20-round DAG:
653        // - Rounds 6-8: authority 0 skips proposals
654        // - Round 11: authority 1 equivocates (1 extra block)
655        // - Round 13: authority 2 equivocates (2 extra blocks)
656        let mut dag_builder = DagBuilder::new(context.clone());
657        dag_builder.layers(1..=5).build();
658        dag_builder
659            .layers(6..=8)
660            .authorities(vec![AuthorityIndex::new_for_test(0)])
661            .skip_block()
662            .build();
663        dag_builder.layers(9..=10).build();
664        dag_builder
665            .layers(11..=11)
666            .authorities(vec![AuthorityIndex::new_for_test(1)])
667            .equivocate(1)
668            .build();
669        dag_builder.layers(12..=12).build();
670        dag_builder
671            .layers(13..=13)
672            .authorities(vec![AuthorityIndex::new_for_test(2)])
673            .equivocate(2)
674            .build();
675        dag_builder.layers(14..=20).build();
676
677        let mut commits = dag_builder
678            .get_sub_dag_and_commits(1..=20)
679            .into_iter()
680            .map(|(_subdag, commit)| commit)
681            .collect::<Vec<_>>();
682
683        // Accept blocks+commits for first 10 rounds
684        let temp_commits = commits.split_off(9);
685        dag_state.accept_block_headers(dag_builder.block_headers(1..=10), DataSource::Test);
686        for commit in commits {
687            dag_state.add_commit(commit);
688        }
689
690        // Metrics should be zero before flush
691        let scoring = dag_state.misbehavior_store();
692        assert_eq!(scoring.persisted_equivocations(), vec![0; committee_size]);
693        assert_eq!(
694            scoring.persisted_missing_proposals(),
695            vec![0; committee_size]
696        );
697
698        // Flush — this triggers misbehavior_counts_to_write
699        dag_state.flush();
700
701        // After flush: authority 0 should have missing proposals in the in-memory
702        // window (cached rounds 6-8 where it didn't propose).
703        // No equivocations yet (those are in rounds 11+, not accepted yet).
704        let scoring = dag_state.misbehavior_store();
705        assert_eq!(scoring.persisted_equivocations(), vec![0; committee_size]);
706        assert!(scoring.in_memory_missing_proposals()[0] > 0);
707
708        // Drop and recover from storage
709        let persisted_missing_before = scoring.persisted_missing_proposals();
710        let in_memory_missing_before_drop = scoring.in_memory_missing_proposals();
711        drop(dag_state);
712        let mut dag_state = DagState::new(context, store);
713
714        // Persisted metrics should be restored, and in-memory recomputed from
715        // the cached block refs loaded during recovery.
716        let scoring = dag_state.misbehavior_store();
717        assert_eq!(
718            scoring.persisted_missing_proposals(),
719            persisted_missing_before
720        );
721        // In-memory is recomputed from cached refs on init (not zero).
722        let in_memory_after_recovery = scoring.in_memory_missing_proposals();
723        assert_eq!(
724            in_memory_after_recovery[0], in_memory_missing_before_drop[0],
725            "In-memory should be recomputed from cached refs on startup"
726        );
727
728        // Accept rounds 11-20 and flush
729        dag_state.accept_block_headers(dag_builder.block_headers(11..=20), DataSource::Test);
730        for commit in temp_commits {
731            dag_state.add_commit(commit);
732        }
733        dag_state.flush();
734
735        // Now equivocations should be tracked
736        let scoring = dag_state.misbehavior_store();
737        let total_eq: u64 = scoring.persisted_equivocations().iter().sum::<u64>()
738            + scoring.in_memory_equivocations().iter().sum::<u64>();
739        assert!(total_eq > 0, "Should have detected equivocations");
740    }

888        let committee_size = 4;

892        let a1 = AuthorityIndex::new_for_test(1);
893        let provable = ConsensusError::TooManyAncestors(10, 5);
894
895        // Seed in_memory provable counts for authority 0 (2) and 1 (1).
896        store.record_faulty_block_header(a0, a0, &provable);
897        store.record_faulty_block_header(a0, a0, &provable);
898        store.record_faulty_block_header(a1, a1, &provable);
899
900        // Flush faulty buffer for authority 0 into persisted; leave authority 1
901        // unflushed so the snapshot must sum across both buckets.
902        let _ =
903            store.update_misbehavior_counts_on_eviction(a0, &BTreeSet::new(), 0, 0, 1, &context);
904
905        // Record 3 more provable faults on authority 0 — these stay in_memory.
906        for _ in 0..3 {
907            store.record_faulty_block_header(a0, a0, &provable);
908        }
909
910        let snapshot = store.snapshot_totals();
911        assert_eq!(snapshot.len(), committee_size);
912        // Authority 0: 2 flushed into persisted + 3 still in_memory = 5 total.
913        // Authority 1: 1 still in_memory (never flushed) = 1 total.
914        // Untouched authorities are zero across both buckets.
915        let provable_totals: Vec<u64> = snapshot
916            .iter()
917            .map(|c| match c {
918                MisbehaviorCounts::V1(v1) => v1.faulty_blocks_provable,
919            })
920            .collect();
921        assert_eq!(provable_totals, vec![5, 1, 0, 0]);
