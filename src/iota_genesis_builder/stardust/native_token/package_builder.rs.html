17const IOTA_FRAMEWORK_GENESIS_REVISION: &str = "framework/genesis/mainnet";

19const MODULE_CONTENT: &str = r#"// Copyright (c) 2024 IOTA Stiftung
20// SPDX-License-Identifier: Apache-2.0
21
22#[allow(lint(share_owned))]
23module 0x0::$MODULE_NAME {
24    use iota::coin;
25    use iota::coin_manager;
26    use iota::url::Url;
27
28    /// The type identifier of coin. The coin will have a type
29    /// tag of kind: `Coin<package_object::$MODULE_NAME::$OTW`
30    /// Make sure that the name of the type matches the module's name.
31    public struct $OTW has drop {}
32
33    /// Module initializer is called once on module publish. A treasury
34    /// cap is sent to the publisher, who then controls minting and burning
35    fun init(witness: $OTW, ctx: &mut TxContext) {
36        let icon_url = $ICON_URL;
37
38        // Create the currency
39        let (mut treasury_cap, metadata) = coin::create_currency<$OTW>(
40            witness,
41            $COIN_DECIMALS,
42            b"$COIN_SYMBOL",
43            $COIN_NAME,
44            $COIN_DESCRIPTION,
45            icon_url,
46            ctx
47        );
48
49        // Mint the tokens and transfer them to the publisher
50        let minted_coins = coin::mint(&mut treasury_cap, $CIRCULATING_SUPPLY, ctx);
51        transfer::public_transfer(minted_coins, ctx.sender());
52
53        // Create a coin manager
54        let (cm_treasury_cap, cm_metadata_cap, mut coin_manager) = coin_manager::new(treasury_cap, metadata, ctx);
55        cm_treasury_cap.enforce_maximum_supply(&mut coin_manager, $MAXIMUM_SUPPLY);
56
57        // Make the metadata immutable
58        cm_metadata_cap.renounce_metadata_ownership(&mut coin_manager);
59
60        // Publicly sharing the `CoinManager` object for convenient usage by anyone interested
61        transfer::public_share_object(coin_manager);
62
63        // Transfer the coin manager treasury capability to the alias address
64        transfer::public_transfer(cm_treasury_cap, iota::address::from_ascii_bytes(&b"$ALIAS"));
65    }
66}
67"#;

69const TOML_CONTENT: &str = r#"[package]
70name = "$PACKAGE_NAME"
71version = "0.0.1"
72edition = "2024"
73
74[dependencies]
75Iota = { git = "https://github.com/iotaledger/iota.git", subdir = "crates/iota-framework/packages/iota-framework", rev = "$GENESIS_REVISION" }
76"#;

78/// Builds and compiles a Stardust native token package.
79pub fn build_and_compile(package: NativeTokenPackageData) -> Result<CompiledPackage> {
80    // Set up a temporary directory to build the native token package
81    let tmp_dir = tempdir()?;
82    let package_path = tmp_dir.path().join("native_token_package");
83    fs::create_dir_all(&package_path).expect("Failed to create native_token_package directory");
84
85    // Write and replace template variables in the Move.toml file
86    write_move_toml(&package_path, &package, IOTA_FRAMEWORK_GENESIS_REVISION)?;
87
88    // Write and replace template variables in the .move file
89    write_native_token_module(&package_path, &package)?;
90
91    // Compile the package
92    move_package::package_hooks::register_package_hooks(Box::new(IotaPackageHooks));
93
94    let build_config = genesis_build_configuration();
95    let compiled_package = build_config.build(&package_path)?;
96
97    // Step 5: Clean up the temporary directory
98    tmp_dir.close()?;
99
100    Ok(compiled_package)
101}

103// Write the Move.toml file with the package name and alias address.
104fn write_move_toml(
105    package_path: &Path,
106    package: &NativeTokenPackageData,
107    iota_framework_genesis_revision: &str,
108) -> Result<()> {
109    let cargo_toml_path = package_path.join("Move.toml");
110    let new_contents = TOML_CONTENT
111        .replace("$PACKAGE_NAME", package.package_name())
112        .replace("$GENESIS_REVISION", iota_framework_genesis_revision);
113    fs::write(&cargo_toml_path, new_contents)?;
114
115    Ok(())
116}

118// Replaces template variables in the .move file with the actual values.
119fn write_native_token_module(package_path: &Path, package: &NativeTokenPackageData) -> Result<()> {
120    let move_source_path = package_path.join("sources");
121    fs::create_dir_all(&move_source_path).expect("Failed to create sources directory");
122    let new_move_file_name = format!("{}.move", package.module().module_name);
123    let new_move_file_path = move_source_path.join(new_move_file_name);
124
125    let icon_url = match &package.module().icon_url {
126        Some(url) => format!(
127            "option::some<Url>(iota::url::new_unsafe_from_bytes({}))",
128            format_string_as_move_vector(url.as_str())
129        ),
130        None => "option::none<Url>()".to_string(),
131    };
132
133    let new_contents = MODULE_CONTENT
134        .replace("$MODULE_NAME", &package.module().module_name)
135        .replace("$OTW", &package.module().otw_name)
136        .replace("$COIN_DECIMALS", &package.module().decimals.to_string())
137        .replace("$COIN_SYMBOL", &package.module().symbol)
138        .replace(
139            "$CIRCULATING_SUPPLY",
140            &package.module().circulating_supply.to_string(),
141        )
142        .replace(
143            "$MAXIMUM_SUPPLY",
144            &package.module().maximum_supply.to_string(),
145        )
146        .replace(
147            "$COIN_NAME",
148            format_string_as_move_vector(package.module().coin_name.as_str()).as_str(),
149        )
150        .replace(
151            "$COIN_DESCRIPTION",
152            format_string_as_move_vector(package.module().coin_description.as_str()).as_str(),
153        )
154        .replace("$ICON_URL", &icon_url)
155        .replace(
156            "$ALIAS",
157            // Remove the "0x" prefix
158            &package.module().alias_address.to_string().replace("0x", ""),
159        );
160
161    fs::write(&new_move_file_path, new_contents)?;
162
163    Ok(())
164}

166/// Converts a string x to a string y representing the bytes of x as hexadecimal
167/// values, which can be used as a piece of Move code.
168///
169/// Example: It converts "abc" to "vector<u8>[0x61, 0x62, 0x63]" plus the
170/// original human-readable string in a comment.
171fn format_string_as_move_vector(string: &str) -> String {
172    let mut byte_string = String::new();
173    byte_string.push_str("/* The utf-8 bytes of '");
174    byte_string.push_str(string);
175    byte_string.push_str("' */\n");
176
177    byte_string.push_str("            vector<u8>[");
178
179    for (idx, byte) in string.as_bytes().iter().enumerate() {
180        byte_string.push_str(&format!("{byte:#x}"));
181
182        if idx != string.len() - 1 {
183            byte_string.push_str(", ");
184        }
185    }
186
187    byte_string.push(']');
188
189    byte_string
190}

192/// Construct the [BuildConfig] for genesis builder
193///
194/// All the configurations are explicitly specified, regardless
195/// of their verbosity so that when any underlying configuration struct is
196/// changed the developer may observe a build error and be able to appropriately
197/// decide which setting should be used here.
198/// In addition we do not rely on silent changes stemming for default
199/// settings being changed erroneously.
200fn genesis_build_configuration() -> BuildConfig {
201    let config = MoveBuildConfig {
202        default_flavor: Some(move_compiler::editions::Flavor::Iota),
203        dev_mode: false,
204        test_mode: false,
205        generate_docs: false,
206        docgen_flags: DocgenFlags::default(),
207        save_disassembly: false,
208        install_dir: None,
209        force_recompilation: false,
210        lock_file: None,
211        fetch_deps_only: false,
212        skip_fetch_latest_git_deps: false,
213        default_edition: None,
214        deps_as_root: false,
215        silence_warnings: false,
216        warnings_are_errors: false,
217        json_errors: false,
218        additional_named_addresses: BTreeMap::default(),
219        lint_flag: LintFlag::LEVEL_DEFAULT,
220        implicit_dependencies: BTreeMap::default(),
221    };
222    BuildConfig {
223        config,
224        run_bytecode_verifier: true,
225        print_diags_to_stderr: false,
226        chain_id: None,
227    }
228}

230#[cfg(test)]
231mod tests {
232
233    use super::*;
234
235    #[test]
236    fn string_to_move_vector() {
237        let tests = [
238            ("", "vector<u8>[]"),
239            ("a", "vector<u8>[0x61]"),
240            ("ab", "vector<u8>[0x61, 0x62]"),
241            ("abc", "vector<u8>[0x61, 0x62, 0x63]"),
242            (
243                "\nöäü",
244                "vector<u8>[0xa, 0xc3, 0xb6, 0xc3, 0xa4, 0xc3, 0xbc]",
245            ),
246        ];
247
248        for (test_input, expected_result) in tests {
249            let move_string = format_string_as_move_vector(test_input);
250            // Ignore the comment and whitespace.
251            let actual_result = move_string.split('\n').next_back().unwrap().trim_start();
252            assert_eq!(expected_result, actual_result);
253        }
254    }
