7use serde::{Deserialize, Serialize};

9/// Operational configurations of a consensus authority.
10///
11/// All fields should tolerate inconsistencies among authorities, without
12/// affecting safety of the protocol. Otherwise, they need to be part of IOTA
13/// protocol config or epoch state on-chain.
14///
15/// NOTE: fields with default values are specified in the serde default
16/// functions. Most operators should not need to specify any field, except
17/// db_path.
18#[derive(Clone, Debug, Deserialize, Serialize)]
19pub struct Parameters {
20    /// Path to consensus DB for this epoch. Required when initializing
21    /// consensus. This is calculated based on user configuration for base
22    /// directory.
23    #[serde(skip)]
24    pub db_path: PathBuf,
25
26    /// Time to wait for parent round leader before sealing a block, from when
27    /// parent round has a quorum.
28    #[serde(default = "Parameters::default_leader_timeout")]
29    pub leader_timeout: Duration,
30
31    /// Sustained spacing between own blocks: long-run production never exceeds
32    /// one block per `min_block_delay`. This avoids generating too many rounds
33    /// when latency is low. This is especially necessary for tests running
34    /// locally. If setting a non-default value, it should be set low enough
35    /// to avoid reducing round rate and increasing latency in realistic and
36    /// distributed configurations.
37    #[serde(default = "Parameters::default_min_block_delay")]
38    pub min_block_delay: Duration,
39
40    /// Soft counterpart of `leader_timeout`: after this duration we are
41    /// willing to propose a block even without a strong-vote quorum, to avoid
42    /// liveness stalls when leader data is slow to propagate. Fires earlier
43    /// than `leader_timeout` and does not force block creation on its own.
44    #[serde(default = "Parameters::default_soft_leader_timeout")]
45    pub soft_leader_timeout: Duration,
46
47    /// Window bounding own block production together with `min_block_delay`:
48    /// idle time accrues budget for bursts of up to `block_rate_window /
49    /// min_block_delay` back-to-back blocks, letting a validator that fell
50    /// behind catch up on rounds instead of skipping them. Set at or below
51    /// `min_block_delay` to disable bursting (fixed spacing between blocks).
52    #[serde(default = "Parameters::default_block_rate_window")]
53    pub block_rate_window: Duration,
54
55    /// Number of block headers to fetch per commit sync request.
56    #[serde(default = "Parameters::default_max_headers_per_commit_sync_fetch")]
57    pub max_headers_per_commit_sync_fetch: usize,
58
59    /// Number of transactions to fetch per commit sync request.
60    #[serde(default = "Parameters::default_max_transactions_per_commit_sync_fetch")]
61    pub max_transactions_per_commit_sync_fetch: usize,
62
63    /// Number of block headers to fetch per header sync (periodic or live)
64    /// request.
65    #[serde(default = "Parameters::default_max_headers_per_header_sync_fetch")]
66    pub max_headers_per_header_sync_fetch: usize,
67
68    /// Number of transactions to fetch per transaction sync request.
69    #[serde(default = "Parameters::default_max_transactions_per_transaction_sync_fetch")]
70    pub max_transactions_per_transaction_sync_fetch: usize,
71
72    /// Time to wait during node start up until the node has synced the last
73    /// proposed block via the network peers. When set to `0` the sync
74    /// mechanism is disabled. This property is meant to be used for amnesia
75    /// recovery.
76    #[serde(default = "Parameters::default_sync_last_known_own_block_timeout")]
77    pub sync_last_known_own_block_timeout: Duration,
78
79    /// The number of rounds of blocks to be kept in the Dag state cache per
80    /// authority. The larger the number the more the blocks that will be
81    /// kept in memory allowing minimising any potential disk access.
82    /// Value should be at minimum 50 rounds to ensure node performance, but
83    /// being too large can be expensive in memory usage.
84    #[serde(default = "Parameters::default_dag_state_cached_rounds")]
85    pub dag_state_cached_rounds: u32,
86
87    /// Rounds a header from a far-future-bounded source may lead the locally
88    /// accepted frontier, in addition to `dag_state_cached_rounds`, before it
89    /// is dropped as too far ahead to connect.
90    #[serde(default = "Parameters::default_peer_round_ahead_margin")]
91    pub peer_round_ahead_margin: u32,
92
93    // Number of authorities commit syncer fetches in parallel.
94    // Both commits in a range and blocks referenced by the commits are fetched per authority.
95    #[serde(default = "Parameters::default_commit_sync_parallel_fetches")]
96    pub commit_sync_parallel_fetches: usize,
97
98    // Number of commits to fetch in a batch, also the maximum number of commits returned per
99    // fetch. If this value is set too small, fetching becomes inefficient.
100    // If this value is set too large, it can result in load imbalance and stragglers.
101    #[serde(default = "Parameters::default_commit_sync_batch_size")]
102    pub commit_sync_batch_size: u32,
103
104    // This affects the maximum number of commit batches being fetched, and those fetched but not
105    // processed as consensus output, before throttling of outgoing commit fetches starts.
106    #[serde(default = "Parameters::default_commit_sync_batches_ahead")]
107    pub commit_sync_batches_ahead: usize,
108
109    /// Maximum number of headers to be included in a bundle. Headers exceeding
110    /// the max allowed limit will be truncated.
111    #[serde(default = "Parameters::default_max_headers_per_bundle")]
112    pub max_headers_per_bundle: usize,
113
114    /// Maximum number of transaction shards to be included in a bundle. Shards
115    /// exceeding the max allowed limit will be truncated.
116    #[serde(default = "Parameters::default_max_shards_per_bundle")]
117    pub max_shards_per_bundle: usize,
118
119    /// Tonic network settings.
120    #[serde(default = "TonicParameters::default")]
121    pub tonic: TonicParameters,
122
123    // Number of commits to fetch in a batch for fast commit syncer, also the maximum number of
124    // commits returned per fetch. If this value is set too small, fetching becomes
125    // inefficient. If this value is set too large, it can result in load imbalance and
126    // stragglers.
127    #[serde(default = "Parameters::default_fast_commit_sync_batch_size")]
128    pub fast_commit_sync_batch_size: u32,
129
130    // Gap threshold for switching between commit syncers. When the gap between quorum and local
131    // commit index is larger than this threshold, FastCommitSyncer fetches. Otherwise,
132    // CommitSyncer fetches.
133    #[serde(default = "Parameters::default_commit_sync_gap_threshold")]
134    pub commit_sync_gap_threshold: u32,
135
136    /// Enable FastCommitSyncer for faster recovery from large commit gaps.
137    /// This is a local node configuration that works in conjunction with the
138    /// protocol-level consensus_fast_commit_sync feature flag. Both must be
139    /// enabled for FastCommitSyncer to run. The protocol flag controls
140    /// whether gRPC endpoints are available, while this local flag controls
141    /// whether this specific node creates and runs the FastCommitSyncer.
142    /// Enabled by default; operators can disable it locally if bugs are
143    /// discovered, without affecting protocol-level endpoint availability.
144    #[serde(default = "Parameters::default_enable_fast_commit_syncer")]
145    pub enable_fast_commit_syncer: bool,
146
147    /// Enable adaptive acknowledgment filtering for StarfishSpeed.
148    /// Local heuristic that drops acks for authorities persistently blamed
149    /// by recent strong-vote masks. Effective only when the protocol-level
150    /// `consensus_starfish_speed` flag is also on. Enabled by default;
151    /// operators can disable it locally without a protocol change.
152    #[serde(default = "Parameters::default_enable_starfish_speed_adaptive_acknowledgments")]
153    pub enable_starfish_speed_adaptive_acknowledgments: bool,
154
155    /// Port for the DAG visualizer gRPC server (localhost only).
156    /// When set, starts a debugging server for real-time DAG visualization.
157    /// Only has an effect when the `dag-visualizer` feature is compiled in.
158    /// Disabled by default (None).
159    #[serde(default)]
160    pub dag_visualizer_port: Option<u16>,
161}

163impl Parameters {
164    pub(crate) fn default_leader_timeout() -> Duration {
165        Duration::from_millis(200)
166    }
167
168    pub(crate) fn default_min_block_delay() -> Duration {
169        if cfg!(msim) || std::env::var("__TEST_ONLY_CONSENSUS_USE_LONG_MIN_BLOCK_DELAY").is_ok() {
170            // Checkpoint building and execution cannot keep up with high commit rate in
171            // simtests, leading to long reconfiguration delays. This is because
172            // simtest is single threaded, and spending too much time in
173            // consensus can lead to starvation elsewhere.
174            Duration::from_millis(400)
175        } else if cfg!(test) {
176            // Avoid excessive CPU, data and logs in tests.
177            Duration::from_millis(250)
178        } else {
179            // For production, use min delay between block being set to 50ms, reducing the
180            // block rate to 20 blocks/sec
181            Duration::from_millis(50)
182        }
183    }
184
185    pub(crate) fn default_soft_leader_timeout() -> Duration {
186        Duration::from_millis(100)
187    }
188
189    pub(crate) fn default_block_rate_window() -> Duration {
190        Duration::from_secs(2)
191    }
192
193    /// Burst capacity: maximum number of own blocks within `block_rate_window`
194    /// (40 in production, 5 in msim, 8 in tests with the default window).
195    pub fn block_rate_burst(&self) -> u64 {
196        let interval_ms = self.min_block_delay.as_millis().max(1) as u64;
197        (self.block_rate_window.as_millis() as u64 / interval_ms).max(1)
198    }
199
200    /// Highest round a header from a far-future-bounded source may have,
201    /// relative to the accepted `frontier`, to still be close enough to
202    /// connect; headers above this are too far ahead and dropped.
203    pub fn far_future_round_ceiling(&self, frontier: u32) -> u32 {
204        frontier
205            .saturating_add(self.dag_state_cached_rounds)
206            .saturating_add(self.peer_round_ahead_margin)
207    }
208
209    /// Maximum number of block headers served per fetch request, depending on
210    /// whether the request comes from commit sync or the header synchronizer.
211    pub fn max_headers_per_fetch(&self, commit_sync: bool) -> usize {
212        if commit_sync {
213            self.max_headers_per_commit_sync_fetch
214        } else {
215            self.max_headers_per_header_sync_fetch
216        }
217    }
218
219    // Maximum number of block headers to fetch per commit sync request.
220    pub(crate) fn default_max_headers_per_commit_sync_fetch() -> usize {
221        if cfg!(msim) {
222            // Exercise hitting blocks per fetch limit.
223            10
224        } else {
225            1000
226        }
227    }
228
229    // Maximum number of transactions to fetch per commit sync request.
230    pub(crate) fn default_max_transactions_per_commit_sync_fetch() -> usize {
231        if cfg!(msim) {
232            // Exercise hitting transactions per fetch limit.
233            10
234        } else {
235            1000
236        }
237    }
238
239    // Maximum number of block headers to fetch per header sync (periodic or
240    // live) request.
241    pub(crate) fn default_max_headers_per_header_sync_fetch() -> usize {
242        if cfg!(msim) {
243            // Exercise hitting blocks per fetch limit.
244            10
245        } else {
246            // TODO: This might should match the value of block headers in the bundle.
247            100
248        }
249    }
250
251    // Maximum number of transactions to fetch per transaction sync request.
252    pub(crate) fn default_max_transactions_per_transaction_sync_fetch() -> usize {
253        if cfg!(msim) { 10 } else { 1000 }
254    }
255
256    pub(crate) fn default_sync_last_known_own_block_timeout() -> Duration {
257        if cfg!(msim) {
258            Duration::from_millis(500)
259        } else {
260            // Here we prioritise liveness over the complete de-risking of block
261            // equivocation. 5 seconds in the majority of cases should be good
262            // enough for this given a healthy network.
263            Duration::from_secs(5)
264        }
265    }
266
267    pub(crate) fn default_dag_state_cached_rounds() -> u32 {
268        if cfg!(msim) {
269            // Exercise reading blocks from store.
270            5
271        } else {
272            500
273        }
274    }
275
276    pub(crate) fn default_peer_round_ahead_margin() -> u32 {
277        1000
278    }
279
280    pub(crate) fn default_commit_sync_parallel_fetches() -> usize {
281        8
282    }
283
284    pub(crate) fn default_commit_sync_batch_size() -> u32 {
285        if cfg!(msim) {
286            // Exercise commit sync.
287            5
288        } else {
289            100
290        }
291    }
292
293    pub(crate) fn default_commit_sync_batches_ahead() -> usize {
294        // This is set to be a multiple of default commit_sync_parallel_fetches to allow
295        // fetching ahead, while keeping the total number of inflight fetches
296        // and unprocessed fetched commits limited.
297        32
298    }
299
300    pub(crate) fn default_max_headers_per_bundle() -> usize {
301        150
302    }
303
304    pub(crate) fn default_max_shards_per_bundle() -> usize {
305        150
306    }
307
308    pub(crate) fn default_fast_commit_sync_batch_size() -> u32 {
309        if cfg!(msim) {
310            // Exercise fast commit sync.
311            5
312        } else {
313            // With ~10KB per commit and 4MB max message size, 1000 commits (~10MB) requires
314            // chunking. The server will chunk commits across multiple response messages.
315            1000
316        }
317    }
318
319    pub(crate) fn default_commit_sync_gap_threshold() -> u32 {
320        if cfg!(msim) {
321            // Use smaller threshold for testing.
322            10
323        } else {
324            // When gap > 1000, FastCommitSyncer is more efficient.
325            // When gap <= 1000, CommitSyncer handles incremental sync.
326            1000
327        }
328    }
329
330    pub(crate) fn default_enable_fast_commit_syncer() -> bool {
331        // Enabled by default. Operators can disable it locally if bugs are discovered,
332        // without waiting for a protocol upgrade.
333        true
334    }
335
336    pub(crate) fn default_enable_starfish_speed_adaptive_acknowledgments() -> bool {
337        true
338    }
339}

377#[derive(Clone, Debug, Deserialize, Serialize)]
378pub struct TonicParameters {
379    /// Keepalive interval and timeouts for both client and server.
380    ///
381    /// If unspecified, this will default to 5s.
382    #[serde(default = "TonicParameters::default_keepalive_interval")]
383    pub keepalive_interval: Duration,
384
385    /// Size of various per-connection buffers.
386    ///
387    /// If unspecified, this will default to 32MiB.
388    #[serde(default = "TonicParameters::default_connection_buffer_size")]
389    pub connection_buffer_size: usize,
390
391    /// Messages over this size threshold will increment a counter.
392    ///
393    /// If unspecified, this will default to 16MiB.
394    #[serde(default = "TonicParameters::default_excessive_message_size")]
395    pub excessive_message_size: usize,
396
397    /// Hard message size limit for both requests and responses.
398    /// This value is higher than strictly necessary, to allow overheads.
399    /// Message size targets and soft limits are computed based on this value.
400    ///
401    /// If unspecified, this will default to 1GiB.
402    #[serde(default = "TonicParameters::default_message_size_limit")]
403    pub message_size_limit: usize,
404
405    /// Maximum number of concurrent HTTP/2 streams a peer may open on a single
406    /// connection. Bounds per-connection request fan-out.
407    ///
408    /// `0` (the default) disables the limit, leaving the transport default.
409    #[serde(default)]
410    pub max_concurrent_streams: u32,
411
412    /// Server-side fallback deadline for requests that omit a `grpc-timeout`
413    /// header. The long-lived block-subscription stream is always exempt.
414    ///
415    /// A zero duration (the default) disables the fallback deadline.
416    #[serde(default)]
417    pub request_timeout: Duration,
418
419    /// Hard size limit for inbound (decoded) requests. Consensus requests are
420    /// small (ref lists); large payloads belong to responses, bounded by
421    /// `message_size_limit`. A smaller inbound bound shrinks the memory a
422    /// single in-flight request can pin before its handler runs.
423    ///
424    /// `0` (the default) falls back to `message_size_limit`.
425    #[serde(default)]
426    pub max_inbound_message_size: usize,
427
428    /// Per-peer, per-RPC admission caps for the inbound consensus server.
429    #[serde(default)]
430    pub admission: AdmissionParameters,
431}

433impl TonicParameters {
434    fn default_keepalive_interval() -> Duration {
435        Duration::from_secs(5)
436    }
437
438    fn default_connection_buffer_size() -> usize {
439        32 << 20
440    }
441
442    fn default_excessive_message_size() -> usize {
443        16 << 20
444    }
445
446    fn default_message_size_limit() -> usize {
447        64 << 20
448    }
449
450    /// Overrides only the inbound resource bounds with the protective preset
451    /// (sized for ~100-validator committees), preserving any
452    /// operator-configured transport settings (keepalive, buffers,
453    /// `message_size_limit`). `default()` stays inert, so the bounds never
454    /// change behaviour unless explicitly enabled.
455    pub fn apply_protective(&mut self) {
456        self.max_concurrent_streams = 64;
457        self.request_timeout = Duration::from_secs(120);
458        self.max_inbound_message_size = 1 << 20;
459        self.admission = AdmissionParameters::protective();
460    }
461
462    /// The inert defaults with the protective bounds applied.
463    pub fn protective() -> Self {
464        let mut params = Self::default();
465        params.apply_protective();
466        params
467    }
468}

485/// Per-peer, per-RPC concurrency caps for the inbound consensus gRPC server.
486///
487/// Each cap bounds how many concurrent requests of one RPC group a single
488/// committee peer (keyed on its authenticated authority index) may have in
489/// flight; a peer cannot consume another peer's budget. These are local,
490/// non-protocol parameters — heterogeneous values across authorities are safe,
491/// so they can be rolled out and tuned per node.
492///
493/// `0` (the default for every cap) disables admission for that group, leaving
494/// the mechanism inert until an operator opts in. Recommended values for the
495/// opt-in protective preset, sized for ~100-validator committees and the local
496/// synchronizer fan-out toward one server: subscriptions 2, header fetches 32,
497/// transaction fetches 16, commit fetches `commit_sync_parallel_fetches` (8).
498#[derive(Clone, Debug, Default, Deserialize, Serialize)]
499pub struct AdmissionParameters {
500    /// Max concurrent block-subscription streams per peer.
501    #[serde(default)]
502    pub max_subscriptions_per_peer: u32,
503
504    /// Max concurrent header fetches per peer
505    /// (`fetch_block_headers` + `fetch_latest_block_headers`).
506    #[serde(default)]
507    pub max_header_fetches_per_peer: u32,
508
509    /// Max concurrent transaction fetches per peer (`fetch_transactions`).
510    #[serde(default)]
511    pub max_transaction_fetches_per_peer: u32,
512
513    /// Max concurrent commit fetches per peer
514    /// (`fetch_commits` + `fetch_commits_and_transactions`).
515    #[serde(default)]
516    pub max_commit_fetches_per_peer: u32,
517}
518
519impl AdmissionParameters {
520    /// Opt-in preset sized for ~100-validator committees and the local
521    /// synchronizer fan-out toward one server.
522    pub fn protective() -> Self {
523        Self {
524            max_subscriptions_per_peer: 2,
525            max_header_fetches_per_peer: 32,
526            max_transaction_fetches_per_peer: 16,
527            max_commit_fetches_per_peer: Parameters::default_commit_sync_parallel_fetches() as u32,
528        }
529    }
