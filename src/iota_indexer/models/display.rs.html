5use diesel::prelude::*;
6use iota_sdk_types::{Event, TypeTag};
7use iota_types::display::DisplayVersionUpdatedEvent;
8
9use crate::schema::display;
10
11#[derive(Queryable, Insertable, Selectable, Debug, Clone)]
12#[diesel(table_name = display)]
13pub struct StoredDisplay {
14    pub object_type: String,
15    pub id: Vec<u8>,
16    pub version: i16,
17    pub bcs: Vec<u8>,
18}
19
20impl StoredDisplay {
21    pub fn try_from_event(event: &Event) -> Option<Self> {
22        if !event.type_.is_display_version_updated() {
23            return None;
24        }
25        let ty = match event.type_.type_params() {
26            [TypeTag::Struct(struct_type)] => struct_type,
27            _ => return None,
28        };
29        let display_event: DisplayVersionUpdatedEvent = bcs::from_bytes(&event.contents).ok()?;
30
31        Some(Self {
32            object_type: ty.to_canonical_string(/* with_prefix */ true),
33            id: display_event.id.bytes.as_bytes().to_vec(),
34            version: display_event.version as i16,
35            bcs: event.contents.clone(),
36        })
37    }
38
39    pub fn to_display_update_event(&self) -> Result<DisplayVersionUpdatedEvent, bcs::Error> {
40        bcs::from_bytes(&self.bcs)
41    }
42}
