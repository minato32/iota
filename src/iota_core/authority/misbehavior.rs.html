12/// Selects which `VersionedMisbehaviorReport` variant peers may submit for
13/// the current epoch. Loaded once from `ProtocolConfig` and threaded through
14/// `MisbehaviorMonitor` / `ReportAggregator` / `Scoreboard` as a `Copy` token.
15///
16/// The schema itself (which categories exist and their layout) lives in
17/// `MisbehaviorObservationsV1`; this enum only versions the wire format and
18/// gates acceptance.
19#[derive(Clone, Copy, Debug, PartialEq, Eq)]
20pub enum MisbehaviorReportVersion {
21    V1,
22}
23
24impl MisbehaviorReportVersion {
25    pub fn from_protocol(protocol_config: &ProtocolConfig) -> Self {
26        match protocol_config.scorer_version_as_option() {
27            None | Some(1) => Self::V1,
28            Some(version) => panic!("Unsupported scorer version {version}"),
29        }
30    }
31
32    /// Returns `true` if the given report's wire format matches this version.
33    pub fn accepts_report(&self, report: &VersionedMisbehaviorReport) -> bool {
34        match self {
35            Self::V1 => matches!(report.payload, MisbehaviorObservations::V1(_)),
36        }
37    }
38}

40/// Per-version zero observations (all metrics, all authorities = 0).
41pub(crate) fn zero_observations(
42    version: MisbehaviorReportVersion,
43    committee_size: usize,
44) -> MisbehaviorObservations {
45    match version {
46        MisbehaviorReportVersion::V1 => MisbehaviorObservations::V1(MisbehaviorObservationsV1 {
47            faulty_blocks_provable: vec![0u64; committee_size],
48            faulty_blocks_unprovable: vec![0u64; committee_size],
49            missing_proposals: vec![0u64; committee_size],
50            equivocations: vec![0u64; committee_size],
51        }),
52    }
53}

55/// Element-wise maximum merge across all metrics. Cross-version merges become
56/// a deliberate design decision when V2 lands (currently impossible — single
57/// variant). Adding a metric to `MisbehaviorObservationsV1` will surface as a
58/// missing-field error here, forcing the new metric to be considered.
59pub(crate) fn merge_max(
60    a: &MisbehaviorObservations,
61    b: &MisbehaviorObservations,
62) -> MisbehaviorObservations {
63    fn elem_max(a: &[u64], b: &[u64]) -> Vec<u64> {
64        a.iter().zip(b.iter()).map(|(x, y)| *x.max(y)).collect()
65    }
66    match (a, b) {
67        (MisbehaviorObservations::V1(x), MisbehaviorObservations::V1(y)) => {
68            MisbehaviorObservations::V1(MisbehaviorObservationsV1 {
69                faulty_blocks_provable: elem_max(
70                    &x.faulty_blocks_provable,
71                    &y.faulty_blocks_provable,
72                ),
73                faulty_blocks_unprovable: elem_max(
74                    &x.faulty_blocks_unprovable,
75                    &y.faulty_blocks_unprovable,
76                ),
77                missing_proposals: elem_max(&x.missing_proposals, &y.missing_proposals),
78                equivocations: elem_max(&x.equivocations, &y.equivocations),
79            })
80        }
81    }
82}

84/// Maps consensus-output counts onto the locally tracked schema. Per-metric
85/// vectors are projected onto `committee_size`: empty (= "not wired by
86/// consensus yet") is silently zero-filled, correct length is used as-is,
87/// any other length is malformed and is logged + zero-filled.
88pub(crate) fn observations_from_consensus_output(
89    counts: ConsensusOutputMisbehaviorCounts,
90    version: MisbehaviorReportVersion,
91    committee_size: usize,
92) -> MisbehaviorObservations {
93    let project = |row: Vec<u64>, name: &'static str| -> Vec<u64> {
94        if row.is_empty() {
95            vec![0u64; committee_size]
96        } else if row.len() == committee_size {
97            row
98        } else {
99            // A non-empty wrong-length row means consensus and iota-core
100            // disagree on committee size — this is a programmer error, not
101            // adversarial input (consensus is local code). Halt in debug /
102            // tests; in production, log loudly and zero-fill so the node
103            // keeps making progress (zero-fill is harmless: merge_max with
104            // zeros leaves prior state untouched).
105            debug_assert_eq!(
106                row.len(),
107                committee_size,
108                "consensus output row for {name} has wrong length — consensus and \
109                 iota-core disagree on committee size"
110            );
111            error!(
112                name,
113                actual = row.len(),
114                expected = committee_size,
115                "consensus output row has wrong length; consensus and iota-core \
116                 disagree on committee size — zero-filling and continuing"
117            );
118            vec![0u64; committee_size]
119        }
120    };
121    match version {
122        MisbehaviorReportVersion::V1 => MisbehaviorObservations::V1(MisbehaviorObservationsV1 {
123            faulty_blocks_provable: project(
124                counts.faulty_blocks_provable,
125                "faulty_blocks_provable",
126            ),
127            faulty_blocks_unprovable: project(
128                counts.faulty_blocks_unprovable,
129                "faulty_blocks_unprovable",
130            ),
131            missing_proposals: project(counts.missing_proposals, "missing_proposals"),
132            equivocations: project(counts.equivocations, "equivocations"),
133        }),
134    }
135}
