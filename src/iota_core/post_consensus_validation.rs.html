4//! Post-consensus validation and owned-object conflict resolution for
5//! `UserTransactionV1` transactions.
6//!
7//! This module merges two formerly separate pipeline stages into a single pass:
8//!
9//! 1. **Semantic validation** — deduplication, already-executed check,
10//!    structural validity, and deny checks (deny lists, gas, ownership, coin
11//!    deny list, Move authenticator).
12//! 2. **Owned-object conflict resolution** (white-flag) — three-tier lock check
13//!    and lock acquisition.
14//!
15//! Processing both in one loop avoids iterating over every transaction twice
16//! and skips expensive validation for transactions that can't acquire object
17//! locks anyway.
18//!
19//! # Per-transaction order within the loop
20//!
21//! 1. Non-`UserTransactionV1` — pass through unchanged.
22//! 2. Dedup by `ConsensusTransactionKey` — silent drop.
23//! 3. Already executed — **retained** as a committee-agreed winner (registers
24//!    its locks, skips re-validation); not dropped. See issue #11649.
25//! 4. `validity_check()` — drop with error.
26//! 5. Three-tier lock conflict check (local HashMap → quarantine → DB) — drop
27//!    with error, except a lock held by the same transaction (a deferred tx's
28//!    own prior-round lock), which is exempt. Cheap; performed before expensive
29//!    checks.
30//! 6. `handle_transaction_validation_checks()` — drop with error. Only reached
31//!    when all locks are free.
32//! 7. All passed — acquire locks in the local tracking map, keep transaction.
33//!
34//! Non-`UserTransactionV1` transactions pass through unchanged.
35

61/// Validates `UserTransactionV1` transactions and resolves owned-object
62/// conflicts in a single pass.
63///
64/// For each `UserTransactionV1` in consensus order:
65/// - Runs deduplication, structural validity, lock conflict check, and deny
66///   checks (deny list, gas, ownership, coin deny list, Move authenticator).
67/// - If all checks pass, acquires owned-object locks in a local tracking map.
68/// - Drops the transaction (with an error) on any failure.
69/// - An already-executed transaction is **retained** (not dropped): it
70///   registers its owned-object locks and skips re-validation. See issue
71///   #11649.
72///
73/// Non-`UserTransactionV1` transactions pass through unchanged.
74///
75/// # Arguments
76///
77/// * `authority_state` — Used for cache reads and deny checks.
78/// * `epoch_store` — Current epoch store (protocol config, lock storage).
79/// * `transactions` — All sequenced transactions for this consensus commit;
80///   modified in-place.
81///
82/// # Returns
83///
84/// `Ok((dropped, locks))` where:
85/// - `dropped` — `(digest, error)` for every semantically-invalid or
86///   lock-conflicting transaction. Silent drops (duplicates) are **not**
87///   included.
88/// - `locks` — Owned-object locks acquired in this commit, to be stored in the
89///   consensus quarantine so subsequent commits can see them.
90pub async fn validate_and_resolve_conflicts(
91    authority_state: &AuthorityState,
92    epoch_store: &Arc<AuthorityPerEpochStore>,
93    transactions: &mut Vec<VerifiedSequencedConsensusTransaction>,
94) -> IotaResult<(
95    Vec<(TransactionDigest, IotaError)>,
96    HashMap<ObjectRef, LockDetails>,
97)> {
98    let mut dropped: Vec<(TransactionDigest, IotaError)> = Vec::new();
99    let mut seen_keys: HashSet<SequencedConsensusTransactionKey> = HashSet::new();
100    // Locks acquired within this commit. Populated for every transaction that
101    // passes all checks; used by subsequent transactions' conflict checks.
102    let mut current_commit_locks: HashMap<ObjectRef, LockDetails> = HashMap::new();
103    // Index-parallel keep flags: true = keep, false = remove.
104    let mut keep = vec![true; transactions.len()];
105
106    for (i, tx) in transactions.iter().enumerate() {
107        // Check #0: Dedup by ConsensusTransactionKey.
108        // The same UserTransactionV1 may appear in DAG blocks from multiple
109        // validators within the same consensus commit. Only the first occurrence
110        // is kept. Silent drop — not added to `dropped`.
111        if !seen_keys.insert(tx.0.key()) {
112            keep[i] = false;
113            continue;
114        }
115
116        // Only validate UserTransactionV1; pass everything else through
117        // unchanged.
118        let transaction = match &tx.0.transaction {
119            SequencedConsensusTransactionKind::External(ConsensusTransaction {
120                kind: ConsensusTransactionKind::UserTransactionV1(t),
121                ..
122            }) => t,
123            _ => continue,
124        };
125
126        let digest = *transaction.digest();
127
128        // Check #1: Already executed (typically by state-sync before this node's
129        // consensus handler reached the commit). It is a committee-agreed winner, so
130        // keep it in the sequence to flow into checkpoint roots like on every other
131        // validator (dropping it forks — issue #11649). Register its owned-object
132        // locks so double-spend siblings still lose, then skip re-validation (#2/#5);
133        // `TransactionManager::enqueue` suppresses the re-execution.
134        if authority_state
135            .get_transaction_cache_reader()
136            .try_is_tx_already_executed(&digest)?
137        {
138            // Byte-based, so safe even though the inputs are already consumed.
139            let owned_inputs = extract_owned_input_objects(tx)?;
140            for obj_ref in &owned_inputs {
141                // A winner cannot be out-locked: an executed tx owns its inputs. A lock
142                // held by a different tx is a consistency violation, not a conflict.
143                if let Some(other) =
144                    find_existing_lock(obj_ref, &current_commit_locks, epoch_store)?
145                {
146                    if other != digest {
147                        fatal!(
148                            "already-executed transaction {:?} has owned input {:?} \
149                             locked by a different transaction {:?}",
150                            digest,
151                            obj_ref,
152                            other,
153                        );
154                    }
155                }
156                current_commit_locks.insert(*obj_ref, digest);
157            }
158            debug!(
159                ?digest,
160                num_owned_inputs = owned_inputs.len(),
161                "Transaction already executed; retained as checkpoint root, skipping re-validation"
162            );
163            // keep[i] stays true so the transaction remains in the sequence.
164            continue;
165        }
166
167        // Check #2: Structural validity.
168        if let Err(e) =
169            transaction.validity_check(epoch_store.protocol_config(), epoch_store.epoch())
170        {
171            warn!(
172                ?digest,
173                error = ?e,
174                "UserTransactionV1 failed validity_check post-consensus, dropping"
175            );
176            dropped.push((digest, e));
177            keep[i] = false;
178            continue;
179        }
180
181        // Check #3: Extract owned input objects for lock conflict detection.
182        let owned_inputs = match extract_owned_input_objects(tx) {
183            Ok(inputs) => inputs,
184            Err(e) => {
185                warn!(
186                    ?digest,
187                    error = ?e,
188                    "Failed to extract owned input objects post-consensus, dropping"
189                );
190                dropped.push((digest, e));
191                keep[i] = false;
192                continue;
193            }
194        };
195
196        // Check #4: Three-tier lock conflict check.
197        // Cheap (HashMap + quarantine + DB lookups); performed before the
198        // expensive deny checks so conflicting transactions are filtered first.
199        //
200        // Locks are keyed by full ObjectRef (id + version + digest), not just
201        // ObjectID. Two transactions referencing the same object at different
202        // versions will NOT conflict here — version freshness is validated
203        // later in Check #5 (deny checks load objects from DB and verify
204        // that the transaction's input refs match the current state).
205        //
206        // Tier 1: Local HashMap (current commit).
207        // Tier 2: Consensus quarantine (previous uncommitted commits).
208        // Tier 3: Persistent DB (committed data).
209        let mut conflict: Option<IotaError> = None;
210        for obj_ref in &owned_inputs {
211            if let Some(locked_by) =
212                find_existing_lock(obj_ref, &current_commit_locks, epoch_store)?
213            {
214                // A lock held by this same transaction is its own lock from a
215                // prior round: it acquired owned-object locks, was then deferred,
216                // and is reloaded this round. A self-held lock is NOT a conflict
217                // - without this guard, the transaction will be dropped with
218                // `ObjectLockConflict` and never executed. Issue #11649 mirrors
219                // this guard for the already-executed branch in Check #1.
220                //
221                // Note that same-commit duplicates cannot slip through here since
222                // Check #0 already dedups by digest, so a same-digest lock here
223                // is always this transaction's own prior-round lock.
224                if locked_by == digest {
225                    continue;
226                }
227
228                debug!(
229                    ?digest,
230                    ?obj_ref,
231                    ?locked_by,
232                    "Transaction conflicts with existing owned-object lock, dropping"
233                );
234                conflict = Some(IotaError::ObjectLockConflict {
235                    obj_ref: *obj_ref,
236                    pending_transaction: locked_by,
237                });
238                break;
239            }
240        }
241        if let Some(e) = conflict {
242            dropped.push((digest, e));
243            keep[i] = false;
244            continue;
245        }
246
247        // Check #5: Deny list, gas, ownership, coin deny list, Move
248        // authenticator. Only reached if all locks are free — skips the
249        // expensive object loading for transactions that would be dropped
250        // by the lock conflict check.
251        //
252        // Safe to skip signature re-verification: the consensus block verifier
253        // (`IotaTxValidator::validate_transactions`) already called
254        // `verify_tx()` on every `UserTransactionV1` before accepting the
255        // block. Re-verifying here would not add safety — if a quorum
256        // committed a bad signature it indicates a protocol-level failure
257        // (2f+1 Byzantine/buggy validators), not something we can recover from
258        // by rejecting the transaction post-consensus. Doing so would also risk
259        // diverging from other honest validators.
260        let verified_tx = VerifiedTransaction::new_from_verified((**transaction).clone());
261        if let Err(e) = authority_state
262            .handle_transaction_validation_checks(&verified_tx, epoch_store)
263            .await
264        {
265            if e.is_storage_or_epoch_error() {
266                return Err(e);
267            }
268            warn!(
269                ?digest,
270                error = ?e,
271                "UserTransactionV1 failed post-consensus deny checks, dropping"
272            );
273            dropped.push((digest, e));
274            keep[i] = false;
275            continue;
276        }
277
278        // All checks passed — acquire owned-object locks in local tracking.
279        let num_owned_inputs = owned_inputs.len();
280        for obj_ref in owned_inputs {
281            current_commit_locks.insert(obj_ref, digest);
282        }
283        debug!(
284            ?digest,
285            num_owned_inputs,
286            "Transaction passed post-consensus validation, acquired all object locks"
287        );
288    }
289
290    if !dropped.is_empty() {
291        warn!(
292            num_dropped = dropped.len(),
293            num_retained = transactions
294                .iter()
295                .enumerate()
296                .filter(|(i, _)| keep[*i])
297                .count(),
298            "Post-consensus validation dropped transactions"
299        );
300    }
301
302    // Remove invalid/duplicate/conflicting entries using the parallel keep
303    // flags.
304    let mut iter = keep.into_iter();
305    transactions.retain(|_| iter.next().unwrap_or(true));
306
307    Ok((dropped, current_commit_locks))
308}

310/// Finds an existing owned-object lock on `obj_ref`, walking three tiers in
311/// order:
312/// 1. `current_commit_locks` — locks acquired earlier in the same commit.
313/// 2. Consensus quarantine — locks from previous uncommitted commits.
314/// 3. Persistent DB — committed locks.
315///
316/// Returns `Ok(Some(locker))` if any tier holds a lock, `Ok(None)` if the
317/// input is free. The caller decides what to do with the result (drop with
318/// `ObjectLockConflict` for a contender, or `fatal!` if a winner is
319/// out-locked).
320fn find_existing_lock(
321    obj_ref: &ObjectRef,
322    current_commit_locks: &HashMap<ObjectRef, LockDetails>,
323    epoch_store: &Arc<AuthorityPerEpochStore>,
324) -> IotaResult<Option<LockDetails>> {
325    if let Some(&locker) = current_commit_locks.get(obj_ref) {
326        return Ok(Some(locker));
327    }
328    if let Some(locker) = epoch_store.get_quarantined_owned_object_lock(obj_ref) {
329        return Ok(Some(locker));
330    }
331    epoch_store.tables()?.get_locked_transaction(obj_ref)
332}

334/// Extracts owned input object references from a `UserTransactionV1`
335/// consensus transaction.
336///
337/// Returns only `ImmOrOwnedMoveObject` inputs (excludes shared objects and
338/// packages) — these are the objects that need lock conflict detection.
339fn extract_owned_input_objects(
340    tx: &VerifiedSequencedConsensusTransaction,
341) -> IotaResult<Vec<ObjectRef>> {
342    let transaction_data = match &tx.0.transaction {
343        SequencedConsensusTransactionKind::External(ConsensusTransaction {
344            kind: ConsensusTransactionKind::UserTransactionV1(transaction),
345            ..
346        }) => transaction.data(),
347        _ => {
348            return Err(IotaError::GenericAuthority {
349                error: "Expected UserTransactionV1 in extract_owned_input_objects".to_string(),
350            });
351        }
352    };
353
354    // Use SenderSignedData::input_objects() rather than
355    // TransactionData::input_objects() to also include any owned objects
356    // that may come from MoveAuthenticator signatures in the future.
357    let owned_objects = transaction_data
358        .input_objects()?
359        .into_iter()
360        .filter_map(|input| match input {
361            InputObjectKind::ImmOrOwnedMoveObject(obj_ref) => Some(obj_ref),
362            InputObjectKind::SharedMoveObject { .. } => None,
363            InputObjectKind::MovePackage(_) => None,
364        })
365        .collect();
366
367    Ok(owned_objects)
368}
