22/// Represents the BigTable tables used by the KvWorker.
23
24// Variants are declared in write order; the BTreeSet<Table> field
25// iterates by derived Ord, which follows declaration order.
26//
27// Order matters: data tables (Objects, Transactions, Checkpoints)
28// are written before their indexes (TransactionsByAddress,
29// CheckpointsByDigest), so a reader that sees an index entry can
30// always resolve the underlying row. Reordering variants will
31// reorder writes and may break that read-after-write invariant.
32#[derive(
33    Debug,
34    Clone,
35    Copy,
36    PartialEq,
37    Eq,
38    Hash,
39    PartialOrd,
40    Ord,
41    Serialize,
42    Deserialize,
43    strum::EnumIter,
44)]
45#[serde(rename_all = "kebab-case")]
46#[non_exhaustive]
47pub enum Table {
48    Objects,
49    Transactions,
50    TransactionsByAddress,
51    Checkpoints,
52    CheckpointsByDigest,
53}
54
55/// This worker implementation is responsible for processing checkpoints by
56/// storing its data as Key-Value pairs. The Key-Value pairs are stored in a
57/// BigTableDB.
58pub struct KvWorker {
59    pub client: BigTableClient,
60    /// The tables enabled for writing by this worker.
61    pub enabled_tables: BTreeSet<Table>,
62}
63
64impl KvWorker {
65    /// Creates a new KvWorker that writes data to all tables.
66    ///
67    /// For selective writing, use [`KvWorker::new_selective`].
68    ///
69    /// # Examples
70    ///
71    /// ```rust
72    /// # use iota_kvstore::KvWorker;
73    /// # use iota_kvstore::{Table, BigTableClient};
74    ///
75    /// # #[tokio::main]
76    /// # async fn main() {
77    /// # std::env::set_var("BIGTABLE_EMULATOR_HOST", "localhost");
78    /// let client = BigTableClient::new_local("instance_id", "column_family")
79    ///     .await
80    ///     .unwrap();
81    ///
82    /// /// Write all available tables to BigTable.
83    /// let worker = KvWorker::new(client);
84    ///
85    /// # drop(worker);
86    /// # }
87    /// ```
88    pub fn new(client: BigTableClient) -> Self {
89        Self {
90            client,
91            // All tables are enabled by default.
92            enabled_tables: Table::iter().collect(),
93        }
94    }
95
96    /// Creates a new KvWorker that writes only to the specified tables.
97    ///
98    /// # NOTE
99    /// Passing an empty iterator yields a worker that performs no writes.
100    /// Use [`KvWorker::new`] to write to all tables.
101    ///
102    /// # Examples
103    ///
104    /// ```rust
105    /// # use iota_kvstore::KvWorker;
106    /// # use iota_kvstore::{Table, BigTableClient};
107    ///
108    /// # #[tokio::main]
109    /// # async fn main() {
110    /// # std::env::set_var("BIGTABLE_EMULATOR_HOST", "localhost");
111    /// let client = BigTableClient::new_local("instance_id", "column_family")
112    ///     .await
113    ///     .unwrap();
114    ///
115    /// /// Write only the `Objects` table to BigTable.
116    /// let worker = KvWorker::new_selective(client, [Table::Objects]);
117    ///
118    /// # drop(worker);
119    /// # }
120    /// ```
121    pub fn new_selective(client: BigTableClient, tables: impl IntoIterator<Item = Table>) -> Self {
122        Self {
123            client,
124            enabled_tables: BTreeSet::from_iter(tables),
125        }
126    }
127}

129#[async_trait]
130impl Worker for KvWorker {
131    type Message = ();
132    type Error = anyhow::Error;
133
134    async fn process_checkpoint(&self, checkpoint: Arc<CheckpointData>) -> anyhow::Result<()> {
135        let mut client = self.client.clone();
136
137        for table in &self.enabled_tables {
138            match table {
139                Table::Objects => {
140                    let objects = checkpoint
141                        .transactions
142                        .iter()
143                        .flat_map(|t| &t.output_objects)
144                        .collect::<Vec<_>>();
145                    client.save_objects(&objects).await?;
146                }
147                Table::Transactions => {
148                    let transactions = checkpoint
149                        .transactions
150                        .iter()
151                        .map(|t| {
152                            TransactionData::new(t, checkpoint.checkpoint_summary.sequence_number)
153                        })
154                        .collect::<Vec<_>>();
155                    client.save_transactions(&transactions).await?;
156                }
157                Table::TransactionsByAddress => {
158                    let entries_by_address = checkpoint
159                        .checkpoint_contents
160                        .enumerate_transactions(&checkpoint.checkpoint_summary)
161                        .zip(&checkpoint.transactions)
162                        .flat_map(|((seq, exec_digest), tx)| {
163                            let digest = exec_digest.transaction;
164                            let tx_data = tx.transaction.transaction_data();
165
166                            let affected = std::iter::once(tx_data.sender())
167                                .chain(std::iter::once(tx_data.gas_owner()))
168                                .chain(tx.effects.all_changed_objects().into_iter().filter_map(
169                                    |(_object_ref, owner, _write_kind)| match owner {
170                                        Owner::Address(a) => Some(a),
171                                        _ => None,
172                                    },
173                                ))
174                                .collect::<HashSet<Address>>();
175
176                            affected
177                                .into_iter()
178                                .map(move |address| (address, seq, digest))
179                        });
180
181                    client
182                        .save_transactions_by_address(entries_by_address)
183                        .await?;
184                }
185                Table::Checkpoints => {
186                    client.save_checkpoint(&checkpoint).await?;
187                }
188                Table::CheckpointsByDigest => {
189                    client.save_checkpoint_by_digest(&checkpoint).await?;
190                }
191            }
192        }
193        Ok(())
194    }
195}
