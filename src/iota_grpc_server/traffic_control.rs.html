4//! Tower [`Layer`] that wires the shared [`TrafficController`] into the
5//! gRPC server.
6//!
7//! The layer extracts the client IP via and calls the traffic controller
8//! with a weight derived from the response's gRPC status code.

10//! [check]: iota_core::traffic_controller::TrafficController::check

61/// Tower [`Service`] wrapper that gates requests through the traffic
62/// controller.
63#[derive(Clone)]
64pub struct TrafficControlService<S> {
65    inner: S,
66    traffic_controller: Arc<TrafficController>,
67    client_id_source: ClientIdSource,
68}
69
70impl<S, ReqBody, ResBody> Service<http::Request<ReqBody>> for TrafficControlService<S>
71where
72    S: Service<http::Request<ReqBody>, Response = http::Response<ResBody>> + Clone + Send + 'static,
73    S::Future: Send + 'static,
74    ReqBody: Send + 'static,
75    ResBody: Default + Send + 'static,
76{
77    type Response = S::Response;
78    type Error = S::Error;
79    type Future = Pin<Box<dyn Future<Output = Result<S::Response, S::Error>> + Send>>;
80
81    fn poll_ready(&mut self, cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
82        self.inner.poll_ready(cx)
83    }
84
85    fn call(&mut self, req: http::Request<ReqBody>) -> Self::Future {
86        let traffic_controller = self.traffic_controller.clone();
87        let remote_addr = req
88            .extensions()
89            .get::<TcpConnectInfo>()
90            .and_then(|info| info.remote_addr());
91        let client = match get_client_ip(req.headers(), remote_addr, &self.client_id_source) {
92            ClientIpStatus::Ok(ip) => Some(ip),
93            other => {
94                warn!(
95                    "Skipping traffic controller request handling for client {remote_addr:?}: {other:?}"
96                );
97                None
98            }
99        };
100
101        // Tower contract: `self.inner` is the ready one (poll_ready set it up),
102        // so call it and leave the clone for the next request.
103        let cloned = self.inner.clone();
104        let mut inner = std::mem::replace(&mut self.inner, cloned);
105
106        Box::pin(async move {
107            if !traffic_controller.check(&client, &None).await {
108                return Ok(Status::resource_exhausted("Too many requests").into_http());
109            }
110
111            let result = inner.call(req).await;
112            if let Ok(response) = &result {
113                let code =
114                    Status::from_header_map(response.headers()).map_or(Code::Ok, |s| s.code());
115                tally(&traffic_controller, client, code);
116            }
117            result
118        })
119    }
120}
121
122fn tally(traffic_controller: &TrafficController, client: Option<IpAddr>, code: Code) {
123    let error_info = if matches!(code, Code::Ok) {
124        None
125    } else {
126        Some((normalize(code), format!("{code:?}")))
127    };
128    traffic_controller.tally(TrafficTally {
129        direct: client,
130        through_fullnode: None,
131        error_info,
132        spam_weight: Weight::one(),
133        timestamp: SystemTime::now(),
134    });
135}

137/// Map a gRPC status code to a tally weight.
138///
139/// Mirrors the conservative shape of the JSON-RPC `normalize`: only obvious
140/// client-side mistakes contribute to the error-policy budget. Server-side
141/// failures (`Internal`, `Unavailable`, `DataLoss`) must not count, or a
142/// flaky backend would auto-block legitimate clients.
143fn normalize(code: Code) -> Weight {
144    match code {
145        Code::InvalidArgument
146        | Code::FailedPrecondition
147        | Code::OutOfRange
148        | Code::Unauthenticated
149        | Code::PermissionDenied => Weight::one(),
150        _ => Weight::zero(),
151    }
